function Test-ValidExecutable {
    param([string]$FilePath)
    
    try {
        if (-not (Test-Path $FilePath)) {
            return $false
        }
        
        $bytes = [System.IO.File]::ReadAllBytes($FilePath)
        
        # Check MZ header
        if ($bytes[0] -ne 0x4D -or $bytes[1] -ne 0x5A) {
            Write-Host "[!] Invalid executable: Missing MZ header" -ForegroundColor Red
            return $false
        }
        
        # Check file size
        $size = $bytes.Length
        if ($size -lt 1024) {
            Write-Host "[!] File too small: $size bytes" -ForegroundColor Red
            return $false
        }
        
        # Check for HTML content (common GitHub error)
        $textStart = [System.Text.Encoding]::UTF8.GetString($bytes, 0, 100)
        if ($textStart -match "<html" -or $textStart -match "404" -or $textStart -match "Not Found") {
            Write-Host "[!] Downloaded HTML instead of executable" -ForegroundColor Red
            return $false
        }
        
        Write-Host "[✓] Valid Windows executable confirmed" -ForegroundColor Green
        return $true
        
    } catch {
        Write-Host "[!] File validation error" -ForegroundColor Red
        return $false
    }
}
